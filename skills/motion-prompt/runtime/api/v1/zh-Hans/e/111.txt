// 色差 / 英文名：Chromatic aberration
// 素材包：photo (摄影效果、光线与风格) · 元素 15 · 类别：镜头与相机效果
// 示例内容：街景被分成红、绿、蓝三个通道，以略有差异的尺寸重新叠加；人物、建筑和路灯边缘出现红色和青色镶边，越靠两侧越明显。
// 动效规格：
//   - 场景 2：街角，建筑、树木和路灯（x 114、292、494）位于 y 132–162 的道路后方；人物站在左前方（脚部位于 x 176、y 200，身高 124 px）。
//   - 分离：图像分别与红色 #FF0000、绿色 #00FF00、蓝色 #0000FF 相乘形成三个副本，再在黑底上相加。
//   - 围绕中心缩放：红色 1 + 2s，绿色 1 + s，蓝色 1，其中 s = 0 → 0.02。
//   - 偏移：左右边缘处红蓝通道最终相距约 11 px，中心处为 0。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合增加老镜头的不完美感、VHS 质感或故障感。
// 标签：色差、彩色边缘、RGB 通道分离、镜头像差、色边、故障效果
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#111
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 15
CELLS[14] = cell((k, u) => {
  const a = env(u), s = 0.02 * a, cx = CW / 2, cy = CH / 2;
  if (s < 0.0005) { stageStreet(k, BASE); return; }
  const [S, ks] = layer('ca'); stageStreet(ks, BASE);
  k.fillStyle = '#000000'; k.fillRect(0, 0, CW, CH);
  for (const [col, sc] of [['#FF0000', 1 + 2 * s], ['#00FF00', 1 + s], ['#0000FF', 1]]) {
    const [L, kl] = layer('ca-' + col); kl.drawImage(S, 0, 0); kl.globalCompositeOperation = 'multiply'; kl.fillStyle = col; kl.fillRect(0, 0, CW, CH); kl.globalCompositeOperation = 'source-over';
    k.save(); k.globalCompositeOperation = 'lighter'; k.translate(cx, cy); k.scale(sc, sc); k.translate(-cx, -cy); k.drawImage(L, 0, 0); k.restore();
  }
});
