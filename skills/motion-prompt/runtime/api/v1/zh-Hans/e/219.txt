// 素材包：diagram (资料中常用的图解与图表名称) · 元素 03 · 类别：整理思路的图
// 示例内容：从左侧的“?”方框开始，树状结构向右按 1 → 2 → 4 分支，以折线连接。
// 动效规格：
//   - 根节点：左侧（x = 82）为 66×50 px 的蓝色方框，内有白色“?”；0.4 s 开始，用 0.3 s 弹性出现（ease-out-back）
//   - 第 2 层：x = 248 处有两个 76×34 px 的中蓝色方框；连接折线在 0.75 s 至 1.15 s 绘出，方框分别在 1.05 s 和 1.13 s 弹性出现
//   - 第 3 层：x = 426 处有四个 92×26 px 的浅蓝色方框；连接线从 1.35 s 开始，间隔 0.06 s，各用 0.4 s 绘出；方框从 1.65 s 开始，间隔 0.08 s 弹性出现
//   - 连接线：灰色、3 px，从父节点右边缘出发，在中途向上或向下折弯，接入子节点左边缘（cubic ease-in-out 绘制）
//   - 结束：整个图形在 4.1 s 至 4.5 s 淡出，卡片保持空白，直到 4.7 s 循环重新开始
// 适用场景：适合完整拆解问题或目标，列出原因或行动。
// 标签：逻辑树、问题树、树状图、结构拆解、互斥且穷尽、问题解决、分支
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 diagram -> ../../kit/diagram.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#219
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 03
function tree(k, w, h, u) {
  bg(k, w, h);
  k.globalAlpha = fade(u);
  const R = [82, h / 2], L2 = [[248, 56], [248, 152]], L3 = [[426, 28], [426, 84], [426, 124], [426, 180]];
  const RW = 66, RH = 50, W2 = 76, H2 = 34, W3 = 92, H3 = 26;
  const elbow = (p, c, pw, cw) => { const x0 = p[0] + pw / 2, x1 = c[0] - cw / 2, xm = (x0 + x1) / 2; return [[x0, p[1]], [xm, p[1]], [xm, c[1]], [x1, c[1]]]; };
  pen(k, C.g4, 3);
  L2.forEach(c => trace(k, elbow(R, c, RW, W2), easeInOut(P(u, 0.75, 1.15))));
  L3.forEach((c, i) => trace(k, elbow(L2[i >> 1], c, W2, W3), easeInOut(P(u, 1.35 + i * 0.06, 1.75 + i * 0.06))));
  const rs = pop(u, 0.4);
  box(k, R[0], R[1], RW, RH, 10, rs, C.blue);
  if (rs > 0.6) txt(k, '?', R[0], R[1] + 2, `700 30px ${FONT}`, '#FFFFFF', 'center', 'middle');
  L2.forEach((c, i) => box(k, c[0], c[1], W2, H2, 9, pop(u, 1.05 + i * 0.08), C.blueM));
  L3.forEach((c, i) => box(k, c[0], c[1], W3, H3, 8, pop(u, 1.65 + i * 0.08), C.blueL));
  k.globalAlpha = 1;
}
CELLS[2] = tree;
