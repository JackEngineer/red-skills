// 光标跟随 / 英文名：Cursor follower
// 素材包：lp (落地页与网页效果) · 元素 11 · 类别：悬停与点击交互
// 示例内容：光标在页面上移动时，一个橙色圆形稍有滞后地跟随。
// 动效规格：
//   - 光标路径：x = 255 + 175·sin(2πt/4.8)，y = 86 + 46·sin(4πt/4.8)（横躺的八字形，每 4.8 s 一圈）。
//   - 圆形半径 17，橙色，不透明度 0.88，绘制在光标下方。
//   - 跟随规则：每帧（60 fps）移动剩余距离的 12%（x += (cursor x − x) × 0.12）。
//   - 箭头光标始终可见。
// 适用场景：适合品牌网站和作品集，增添轻松、有操作感的反馈。
// 标签：光标、自定义光标、光标跟随、鼠标跟随、线性插值、轨迹、交互
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#083
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 11
CELLS[10] = (k, w, h, u) => {
  frame(k);
  const cur = t => [255 + 175 * Math.sin(2 * Math.PI * t / CYC), 86 + 46 * Math.sin(4 * Math.PI * t / CYC)];
  // 按 60 帧每秒计算，每帧走完剩余距离的 12%。
  let t = u - 2, [fx, fy] = cur(t);
  for (let i = 0; i < 120; i++) { t += 1 / 60; const [tx, ty] = cur(t); fx += (tx - fx) * 0.12; fy += (ty - fy) * 0.12; }
  view(k, () => {
    txt(k, '创意工作室', VW / 2, 74, `800 34px ${EN}`, C.ink, 'center', 'alphabetic', '3px');
    bars(k, VW / 2 - 120, 96, 240, 2, C.faint, 6, 12, 0.7);
    btn(k, VW / 2 - 50, 128, 100, 24, '查看作品', C.ink);
    k.fillStyle = C.acc; k.globalAlpha = 0.88; k.beginPath(); k.arc(fx, fy, 17, 0, 7); k.fill(); k.globalAlpha = 1;
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
