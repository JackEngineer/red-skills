//   - 后层波浪为白色、不透明度 0.45，中心 y = 100，由波长 260 px、振幅 8 px 与波长 130 px、振幅 4 px 的两个正弦波相加而成。
//   - 前层波浪为白色，中心 y = 110，由波长 300 px、振幅 9 px 与波长 110 px、振幅 3.5 px 的波相加，接入下方白色区块。
//   - 相位 a = 2π·t/4.7，后层使用 a + 1 和 −2a，前层使用 −a 和 2a + 0.5。每层的两个正弦波反向运动，因此形状持续变化，每 4.7 s 返回起始状态。
// 适用场景：适合希望区块边缘柔和而非笔直的页面，例如海洋、夏日或儿童主题网站。

// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 21
CELLS[20] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const a = 2 * Math.PI * u / CYC;
  const wave = (base, A1, L1, p1, A2, L2, p2) => { k.beginPath(); k.moveTo(0, VH + 1); for (let x = 0; x <= VW + 4; x += 4) k.lineTo(x, base + A1 * Math.sin(2 * Math.PI * x / L1 + p1) + A2 * Math.sin(2 * Math.PI * x / L2 + p2)); k.lineTo(VW, VH + 1); k.closePath(); };
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 120); g.addColorStop(0, '#3B64E8'); g.addColorStop(1, '#8FA9F7'); k.fillStyle = g; k.fillRect(0, 0, VW, VH);
    k.fillStyle = 'rgba(255,255,255,.9)'; k.beginPath(); k.arc(430, 44, 22, 0, 7); k.fill();
    txt(k, '夏日优惠', 34, 42, `800 22px ${JP}`, '#fff'); btn(k, 34, 56, 96, 24, '了解详情', '#fff', C.blue, 11);
    k.fillStyle = 'rgba(255,255,255,.45)'; wave(100, 8, 260, a + 1, 4, 130, -2 * a); k.fill();
    k.fillStyle = '#fff'; wave(110, 9, 300, -a, 3.5, 110, 2 * a + 0.5); k.fill();
    txt(k, '推荐方案', 34, 146, `800 15px ${JP}`, C.ink); bars(k, 170, 140, 290, 1, C.faint, 6);
    bars(k, 34, 156, 200, 1, C.faint, 5);
  });
};
