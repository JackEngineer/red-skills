// 路径文字 / 英文名：Text on a path
// 素材包：motion2 (动态图形（第 2 弹）) · 元素 11 · 类别：文字与图形
// 示例内容：一行文字沿波浪线流动，每个汉字随路径转向。
// 动效规格：
//   - 路径：y = 118 + 42·sin(2πx ÷ 330 + 0.6)，x = −70 至 609；下方 6 px 处以藏青色虚线显示，虚线 14/10、线宽 5。
//   - 重复显示“文字沿着波浪线前行 • ”（42 px，粗体，字距 2 px）。每个字符按弧长定位在路径上，并旋转至路径方向。
//   - 文字每 4.7 s 沿路径前进一个完整字符串的长度，保证循环衔接。
//   - 汉字为白色，圆点为黄色。
// 适用场景：适合让文字沿形状排列和运动，例如环绕标志，沿道路、河流或波浪流动，或形成循环文字带。
// 标签：路径文字、路径文字、弧形文字、文字、跑马灯、波浪、滚动
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion2 -> ../../kit/motion2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#131
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 11
const TOP = (() => { const pts = []; for (let x = -70; x <= CW + 70; x += 4) pts.push([x, 118 + 42 * Math.sin(x * 2 * Math.PI / 330 + 0.6)]); return { pts, cum: pathLen(pts) }; })();
CELLS[10] = (k, w, h, u) => {
  stage(k);
  k.save(); k.setLineDash([14, 10]); k.strokeStyle = C.navy3; k.lineWidth = 5; k.lineCap = 'round'; k.beginPath(); TOP.pts.forEach((p, i) => i ? k.lineTo(p[0], p[1] + 6) : k.moveTo(p[0], p[1] + 6)); k.stroke(); k.restore();
  const f = `800 42px ${JP}`, S = '文字沿着波浪线前行 • '; k.font = f;
  const ws = [...S].map(ch => k.measureText(ch).width + 2), run = ws.reduce((a, b) => a + b, 0), L = TOP.cum[TOP.cum.length - 1], shift = (u / CYC) * run;
  for (let r = -2; r * run + shift < L; r++) {
    let s = r * run + shift;
    [...S].forEach((ch, j) => {
      const c = s + ws[j] / 2; s += ws[j];
      if (c < 0 || c > L || ch === ' ') return;
      const p = ptAt(TOP.pts, TOP.cum, c / L);
      k.save(); k.translate(p[0], p[1]); k.rotate(p[2]); txt(k, ch, 0, 0, f, ch === '•' ? C.yel : '#fff', 'center', 'alphabetic'); k.restore();
    });
  }
};
