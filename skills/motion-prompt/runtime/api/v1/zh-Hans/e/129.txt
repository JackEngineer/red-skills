// 文字乱码效果 / 英文名：Text scramble
// 素材包：motion2 (动态图形（第 2 弹）) · 元素 09 · 类别：文字与图形
// 示例内容：随机字符不断变化，随后从左向右逐个锁定成正确单词。
// 动效规格：
//   - 前 2.35 s 显示“动效进行中”，后 2.35 s 显示“创意进行时”，各五个汉字，字号 80 px。
//   - 尚未锁定的位置随机显示汉字、数字 0–9 和 #$%&@?，每 4.7 s 变化 76 次，约每 0.062 s 一次，薄荷绿，不透明度 75%。
//   - 从左到右锁定：第 i 位在 0.45 + 0.18i s 锁定，最后一位在 1.17 s。新锁定汉字先显示黄色 0.22 s，再变白，并以缩放 1 + 0.2·e^(−14t) 回稳。
//   - 从 1.85 s 开始，各位置自右向左恢复随机字符，第 i 位在 1.85 + 0.08 × (4 − i) s 开始。
//   - 每个位置下方有短条：已锁定为蓝色，下一个待锁定为黄色，其余为藏青色。
// 适用场景：适合让标题、数字或代号以解码、黑客或科技风格出现。
// 标签：文字乱码效果、打乱字符、解码、随机重排、随机、文字、黑客风格
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion2 -> ../../kit/motion2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#129
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 09
const SCR_GL = '春夏秋冬风花雪月山川河海星辰光影云雨霜露0123456789#$%&@?';
CELLS[8] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, word = n ? '创意进行时' : '动效进行中', N = 5;
  const f = `700 80px ${JP}`; k.font = f;
  const cw = k.measureText('动').width + 16, x0 = w / 2 - cw * (N - 1) / 2, Y = h / 2 - 8;
  const step = Math.floor(u / (CYC / 76));
  for (let i = 0; i < N; i++) {
    const tl = 0.45 + i * 0.18, tu = 1.85 + (N - 1 - i) * 0.08, locked = lu >= tl && lu < tu;
    let ch, col, sc = 1;
    if (locked) { ch = word[i]; const age = lu - tl; col = age < 0.22 ? C.yel : '#fff'; sc = 1 + 0.2 * Math.exp(-age * 14); }
    else { ch = SCR_GL[Math.floor(H01(step * 131 + i * 977 + n * 31) * SCR_GL.length)]; col = C.mint; }
    k.save(); k.translate(x0 + i * cw, Y); k.scale(sc, sc); k.globalAlpha = locked ? 1 : 0.75; txt(k, ch, 0, 0, f, col, 'center', 'middle'); k.restore();
    const next = !locked && lu < tl && (i === 0 || lu >= 0.45 + (i - 1) * 0.18);
    k.fillStyle = locked ? C.blue : next ? C.yel : C.navy3; rr(k, x0 + i * cw - cw / 2 + 8, Y + 58, cw - 16, 8, 4); k.fill();
  }
};
