// 追随拍摄 / 英文名：Panning shot
// 素材包：photo (摄影效果、光线与风格) · 元素 12 · 类别：镜头与相机效果
// 示例内容：相机跟随行驶的汽车，汽车在画面中央保持清晰，只有后方的建筑、树木和路灯横向拖出模糊轨迹。
// 动效规格：
//   - 街角：建筑、树木和路灯位于 y 132–162 的道路后方（前景没有较大的人物）。
//   - 汽车：无效果时在 5 s 内从 x −80 行驶到 620；有效果时保持在 x 270 ±10 px。
//   - 拖影：背景横向分布在 46 px 范围内绘制 12 次，再取平均（分布宽度 0 → 46 px）。
//   - 车轮：辐条线每 5 s 转 12 圈。
//   - 右上角快门速度 1/1000 → 1/15。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合在静态照片中表现汽车、自行车或跑步者的速度。
// 标签：追随拍摄、追随拍摄、运动模糊、速度、体育摄影、慢门
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#108
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 12
CELLS[11] = cell((k, u) => {
  const a = env(u);
  const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
  const S = 46 * a, n = S < 0.5 ? 1 : 12;
  for (let i = 0; i < n; i++) {   // 水平平移背景并叠加多次绘制结果，平均得到拖影。
    k.save(); k.globalAlpha = 1 / (i + 1); k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0); stageStreet(k, BASE, { car: false, person: false }); k.restore();
  }
  const D = pen(k); car(D, x, BASE, { head: BASE.lit });
  // 每轮演示车轮恰好转 12 圈，首尾无跳变。
  k.save(); k.strokeStyle = BASE.tire; k.lineWidth = 1.6;
  for (const wx of [x - 21, x + 21]) { const t = u / CYC * TAU * 12; k.beginPath(); k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5); k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5); k.stroke(); }
  k.restore();
  hud(k, a > 0.5 ? '1/15' : '1/1000');
});
