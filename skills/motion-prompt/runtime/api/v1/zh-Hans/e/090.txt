// 懒加载 / 英文名：Lazy loading
// 素材包：lp (落地页与网页效果) · 元素 18 · 类别：加载、导航与循环动效
// 示例内容：灰色占位块滚入视口后，中央先出现旋转加载指示器，图片加载完成后淡入；第一行从开始就已加载。
// 动效规格：
//   - 3 × 3 网格，占位块为 151 × 92；第一行从开始就显示。滚动：0.35–1.1 s 从 0→110 px，1.7–2.5 s 从 110→225 px，3.5–4.5 s 从 225→0（三次方缓入缓出）。
//   - 占位块上边缘越过视口底部（168 px）时开始加载，每列比左侧一列延迟 0.12 s。
//   - 加载指示器在占位块中央旋转 0.45 s，半径 11，每 0.8 s 一圈。
//   - 随后图片在 0.35 s 内从 0→1 淡入（三次方缓出）。尚未加载的占位块显示图片图标。
//   - 从视口底部离开的占位块在不可见处重置。
// 适用场景：适合图片较多的页面，加快首屏加载。
// 标签：懒加载、懒加载、图片、占位内容、性能、加载转圈、图库
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#090
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 18
CELLS[17] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.1, 110], [1.7, 110], [2.5, 225], [3.5, 225], [4.5, 0]], t), y = SY(u);
  const COLX = [15, 180, 345], IW = 151, IH = 92, ROWS = [12, 168, 276], WAIT = 0.45, FADE = 0.35;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '作品集', 15, 140, `800 18px ${JP}`, C.ink); bars(k, 120, 128, 200, 1);
    ROWS.forEach((top, r) => COLX.forEach((x, c) => {
      const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * 0.12;
      k.fillStyle = C.skel; rr(k, x, top, IW, IH, 6); k.fill();
      if (age < WAIT + FADE) {
        k.save(); k.translate(x + IW / 2, top + IH / 2); k.strokeStyle = '#B9C1CF'; k.lineWidth = 2; k.lineJoin = 'round';
        if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue);
        else { rr(k, -15, -11, 30, 22, 3); k.stroke(); k.beginPath(); k.moveTo(-11, 7); k.lineTo(-3, -1); k.lineTo(3, 5); k.lineTo(7, 1); k.lineTo(11, 6); k.stroke(); }
        k.restore();
      }
      if (age >= WAIT) { k.save(); k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE)); pic(k, x, top, IW, IH, (r + c) % 3, 6); k.restore(); }
    }));
    k.restore();
    sbar(k, y / 225, 0.45);
  });
};
