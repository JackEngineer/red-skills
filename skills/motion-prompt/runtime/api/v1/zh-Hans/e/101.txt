// 伦勃朗光 / 英文名：Rembrandt lighting
// 素材包：photo (摄影效果、光线与风格) · 元素 05 · 类别：布光与时段
// 示例内容：侧前方 45°、上方 45° 的灯光亮起；脸部右侧陷入阴影，只在暗侧脸颊上留下一个小小的倒三角光斑。
// 动效规格：
//   - 场景 1：地平线位于 y 150，前方为头肩人像（头部中心 x 270、y 78，半径 29 px，无五官或头发）。背景变暗（天空 #E7E5E1）。
//   - 头部：x = 270 + 0.04·29 右侧为阴影 #33405A，左侧为亮部 #FFC75A。
//   - 三角光斑：三个顶点为 (x 270+0.24R, y 78−0.12R)、(x 270+0.7R, y 78−0.12R)、(x 270+0.24R, y 78+0.5R)，R = 29（约宽 13 px、高 18 px），与亮侧之间隔着约 6 px 宽的阴影带。
//   - 身体：中心 −0.45×56 px 左侧受光，+0.3×56 px 右侧为阴影；俯视图显示灯光位于 45°，右上角读数“45° · ↑45°”。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合具有绘画感、立体感和古典气质的人像。
// 标签：伦勃朗光、人像布光、三角形、45° 侧光、明暗对照法、绘画质感
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#101
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 05
CELLS[4] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    kk.fillStyle = sh; kk.fillRect(PX + 0.04 * R, PY - R - 2, R * 2, R * 2 + 4);
    kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.clip();
    kk.fillStyle = lit; kk.beginPath();
    kk.moveTo(PX + 0.24 * R, PY - 0.12 * R); kk.lineTo(PX + 0.7 * R, PY - 0.12 * R); kk.lineTo(PX + 0.24 * R, PY + 0.5 * R); kk.closePath(); kk.fill();
  });
  bust(k, PX, PY, R, 'rgba(0,0,0,0)', kk => {
    kk.fillStyle = 'rgba(0,0,0,0)';
    torso(kk, PX, PY, R, PX - W, PX - 0.45 * W, lit);
    torso(kk, PX, PY, R, PX + 0.3 * W, PX + W, sh);
  });
  plan(k, [[Math.PI / 4, a]]);
  hud(k, '45° · ↑45°');
});
