// 移轴效果 / 英文名：Tilt-shift
// 素材包：photo (摄影效果、光线与风格) · 元素 10 · 类别：镜头与相机效果
// 示例内容：街景的上下区域变模糊，只有中间包含道路、汽车和小人物的条带保持清晰，让街道看起来像模型。
// 动效规格：
//   - 街角（建筑、树木和路灯位于 y 132–162 的道路后方），不含较大的人物；四名身高 23–26 px 的小人物站在人行道上（脚部位于 y 172）；汽车每 5 s 从 x −60 行驶到 600。
//   - 清晰区域：y 66–176 保持清晰。
//   - 模糊：y 0–42 和 y 190–208 从 0 → 7 px；y 42–66 和 y 176–190 从 0 → 3 px。
//   - 条带边缘有虚线（线段 6 px，间隔 5 px，#8E8E93，不透明度 75%）。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合把城市或活动现场呈现为玩具模型世界。
// 标签：移轴效果、微缩模型、微缩效果、微缩模型效果、选择性对焦、微缩城市、清晰对焦带
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#106
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 10
CELLS[9] = cell((k, u) => {
  const a = env(u), carX = lerp(-60, 600, u / CYC);
  const minis = D => { [[196, 26], [322, 26], [340, 23], [452, 25]].forEach(([x, h]) => figure(D, x, 172, h, BASE.person)); };
  const [S, ks] = layer('tilt'); stageStreet(ks, BASE, { car: carX, person: false, after: minis });
  const [B1, k1] = layer('tilt-1'); k1.filter = `blur(${(3 * a).toFixed(2)}px)`; k1.drawImage(S, 0, 0); k1.filter = 'none';
  const [B2, k2] = layer('tilt-2'); k2.filter = `blur(${(7 * a).toFixed(2)}px)`; k2.drawImage(S, 0, 0); k2.filter = 'none';
  k.drawImage(S, 0, 0);
  const band = (img, y0, y1) => { k.save(); k.beginPath(); k.rect(0, y0, CW, y1 - y0); k.clip(); k.drawImage(img, 0, 0); k.restore(); };
  band(B2, 0, 42); band(B1, 42, 66); band(B1, 176, 190); band(B2, 190, CH);
  k.save(); k.globalAlpha = 0.75 * a; k.strokeStyle = '#8E8E93'; k.lineWidth = 1.4; k.setLineDash([6, 5]);
  k.beginPath(); k.moveTo(0, 66); k.lineTo(CW, 66); k.moveTo(0, 176); k.lineTo(CW, 176); k.stroke(); k.setLineDash([]);
  k.restore();
});
