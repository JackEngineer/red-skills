// 素材包：diagram (资料中常用的图解与图表名称) · 元素 07 · 类别：整理思路的图
// 示例内容：编号 1 至 4 的圆围成环形，箭头依次绘出并闭合，随后蓝色高亮沿环移动一周。
// 动效规格：
//   - 布局：四个浅蓝色圆（半径 26 px，编号 1–4）位于椭圆的上、右、下、左，椭圆横向和纵向半径分别为 128 px、70 px；从 0.4 s 开始，间隔 0.32 s 弹性出现
//   - 箭头：沿椭圆从一个圆连至下一个圆的弧线（中蓝、3 px、头部 13 px），从 0.7 s 开始，间隔 0.32 s，各用 0.28 s 绘出；1.94 s 闭合成环
//   - 高亮：2.2 s 至 3.8 s，实心蓝色填充按 1 → 2 → 3 → 4 移动，每步 0.4 s，边缘柔和
//   - 结束：整个图形在 4.1 s 至 4.5 s 淡出，卡片保持空白，直到 4.7 s 循环重新开始
// 适用场景：适合展示回到起点并持续重复的过程，例如 PDCA。
// 标签：循环、循环图、循环、计划执行检查改进循环、循环流程、环形、迭代
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 diagram -> ../../kit/diagram.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#223
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 07
function cycle(k, w, h, u) {
  bg(k, w, h);
  k.globalAlpha = fade(u);
  const cx = w / 2, cy = h / 2, RX = 128, RY = 70, nr = 26, n = 4;
  const ang = i => -Math.PI / 2 + i * TAU / n, pos = a => [cx + Math.cos(a) * RX, cy + Math.sin(a) * RY];
  for (let i = 0; i < n; i++) {
    const t = easeInOut(P(u, 0.7 + i * 0.32, 0.98 + i * 0.32));
    if (t <= 0) continue;
    const A = pos(ang(i)), B = pos(ang(i + 1)), pts = [];
    for (let j = 0; j <= 60; j++) {
      const q = pos(lerp(ang(i), ang(i + 1), j / 60));
      if (Math.hypot(q[0] - A[0], q[1] - A[1]) > nr + 10 && Math.hypot(q[0] - B[0], q[1] - B[1]) > nr + 8) pts.push(q);
    }
    pen(k, C.blueM, 3); trace(k, pts, t, 13, C.blueM);
  }
  // 高亮从 2.2 秒开始，在 3.8 秒时走完一圈。每步持续 0.4 秒，边缘采用渐变。
  const hp = P(u, 2.2, 3.8) * n;
  for (let i = 0; i < n; i++) {
    const s = pop(u, 0.4 + i * 0.32, 0.28);
    if (s <= 0.01) continue;
    const dd = hp - i;
    const on = (u > 2.15 && u < 4.0 && dd > -0.25 && dd < 1.25) ? clamp(Math.min(dd + 0.25, 1.25 - dd) * 2.5) : 0;
    const [x, y] = pos(ang(i));
    k.save(); k.translate(x, y); k.scale(s, s);
    k.fillStyle = C.blueL; k.beginPath(); k.arc(0, 0, nr, 0, TAU); k.fill();
    if (on > 0) { k.save(); k.globalAlpha *= on; k.fillStyle = C.blue; k.beginPath(); k.arc(0, 0, nr, 0, TAU); k.fill(); k.restore(); }
    txt(k, String(i + 1), 0, 1, `700 22px ${FONT}`, on > 0.5 ? '#FFFFFF' : C.blue, 'center', 'middle');
    k.restore();
  }
  k.globalAlpha = 1;
}
CELLS[6] = cycle;
