// 子弹时间 / 英文名：Bullet time
// 素材包：camera2 (镜头运动与转场（第 2 弹）) · 元素 23 · 类别：转场与剪辑
// 示例内容：人物跳到最高点时，时间几乎停止，相机绕着人物环行。
// 动效规格：
//   - 素材时间：0–1.5 s 正常播放，1.5 s 至 3.3 s 以 0.1× 播放（1.8 s 内只推进 0.18 s 素材），之后恢复正常（素材时间 = u - 1.62）。
//   - 主体：9 m 处的人物在素材时间 0.6–2.6 s 之间沿抛物线跳起（最高 1.1 m，双臂举起）；慢速部分仿佛悬停在最高点附近（进度 45–54%）；九片树叶按相同素材时间下落。
//   - 相机：慢速部分沿人物周围半径 6 m 的圆弧从 -0.6 rad 移至 +0.6 rad（约 69°），三次缓入缓出（约为 cubic-bezier(0.65,0,0.35,1)）；始终朝向圆心，高度 1.6 m，俯仰角 -0.05 rad。
//   - 接下来 4.7 s 反向环行（+0.6 → -0.6 rad），因此 9.4 s 循环无缝（#030 Orbit 不会停止时间）。
//   - 地面有半径 1.3 m 的虚线圆，10 m 处有一圈树木；右上角读数为“×1.0” → “子弹时间 ×0.1” → “×1.0”。
// 适用场景：适合以三维视角展示决定性瞬间，或作为动作场面的高潮。
// 标签：子弹时间、时间切片、瞬间定格、环绕、慢动作、矩阵、特效
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera2 -> ../../kit/camera2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#191
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 23
CELLS[22] = shot((k, w, h, u, cyc) => {
  const T0 = 1.5, T1 = 3.3, SLOW = 0.1;
  const tj = u < T0 ? u : u < T1 ? T0 + (u - T0) * SLOW : u - (T1 - T0) * (1 - SLOW);   // 将现实经过的时间换算为画面内的时间。
  const q = P(tj, 0.6, 2.6), e = eIO(P(u, T0, T1));
  const a = cyc % 2 === 0 ? lerp(-0.6, 0.6, e) : lerp(0.6, -0.6, e), R = 6, P0 = [0, 0, 9];
  const c = cam({ x: P0[0] + Math.sin(a) * R, z: P0[2] - Math.cos(a) * R, yaw: -a, y: 1.6, pitch: -0.05 });
  world(k, w, h, c, {
    person: { z: 9, y: jumpY(q), armUp: q > 0 && q < 1, face: a }, ring: [0, 9, 10], trees: false, extra: leaves(tj, 1.5).map(l => [l[0], l[1], l[2] + 3, l[3]]),
    after: (kk, cc) => { kk.strokeStyle = C.blueM; kk.lineWidth = 2; kk.setLineDash([5, 5]); kk.beginPath(); for (let i = 0; i <= 48; i++) { const b = i / 48 * Math.PI * 2; const qq = proj(cc, Math.sin(b) * 1.3, 0, 9 + Math.cos(b) * 1.3); i ? kk.lineTo(qq.x, qq.y) : kk.moveTo(qq.x, qq.y); } kk.stroke(); kk.setLineDash([]); }
  });
  return u >= T0 && u < T1 ? `子弹时间 ×${SLOW.toFixed(1)}` : '×1.0';
});
