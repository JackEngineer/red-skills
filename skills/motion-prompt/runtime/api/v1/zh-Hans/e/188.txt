// 素材包：camera2 (镜头运动与转场（第 2 弹）) · 元素 20 · 类别：转场与剪辑
// 示例内容：圆形窗口围绕人物逐渐收拢，直到画面全黑，再在下一场景中重新打开。
// 动效规格：
//   - 圆形收拢：1.3 s 至 2.7 s（1.4 s），圆形窗口半径从 340 px 缩至 0（二次缓入缓出，约为 cubic-bezier(0.45,0,0.55,1)）；圆外为黑色（#141418）。
//   - 窗口中心：当前画面的主体投影到屏幕的位置，即 A 中人物的头部（左侧 0.6 m，高 1.63 m，距离 7 m），或 B 中的汽车（右侧 1.2 m，高 0.9 m，距离 9 m）。
//   - 全黑：2.7–3.1 s；2.9 s 时硬切换入下一画面。
//   - 圆形展开：3.1 s 至 4.4 s（1.3 s），圆形围绕下一画面的主体从 0 增至 340 px。
//   - 边缘：半径至少为 1 px 时，用 2 px 蓝色圆线标记边缘；两幅画面中相机均固定。
//   - 每 4.7 s 交替 A→B 和 B→A（往返 9.4 s）；右上角读数为“圆孔收拢” / “圆孔打开”。
// 适用场景：适合默片或卡通风格的结尾，或轻松俏皮的段落间隔。
// 标签：圈出、圈入、光圈划像、圆形、复古胶片、卡通、转场
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera2 -> ../../kit/camera2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#188
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 20
CELLS[19] = shot(abCell((k, w, h, u, A, B) => {
  const out = eIOs(P(u, 1.3, 2.7)), inn = eIOs(P(u, 3.1, 4.4));
  const first = u < 2.9, cur = first ? A : B;
  cur(k, w, h);
  // 遮罩中心跟随当前镜头的主体，A 是人物的头部，B 是汽车。
  const c = cur === sceneA ? proj(cam({ y: 1.4 }), -0.6, 1.63, 7) : proj(cam({ y: 1.6, pitch: 0.1 }), 1.2, 0.9, 9);
  const rad = first ? lerp(340, 0, out) : lerp(0, 340, inn);
  const cx = w / 2 + c.x, cy = h / 2 + c.y;
  if (rad < 339) {
    k.fillStyle = '#141418'; k.beginPath(); k.rect(0, 0, w, h); k.arc(cx, cy, Math.max(0, rad), 0, Math.PI * 2, true); k.fill('evenodd');
    if (rad > 1) { k.strokeStyle = C.blue; k.lineWidth = 2; k.beginPath(); k.arc(cx, cy, rad, 0, 7); k.stroke(); }
  }
}, u => u < 2.9 ? '圆孔收拢' : '圆孔打开'));
