CELLS[23] = shot((k, w, h, u) => {
  let i = 0, t0 = 0; while (i < MT.length - 1 && u >= t0 + MT[i][0]) { t0 += MT[i][0]; i++; }
  const r = P(u, CYC - 0.3, CYC);
  blend(k, kk => MT[i][2](kk, w, h, u - t0), kk => MT[0][2](kk, w, h, 0), eIOs(r));   // 最后 0.3 秒回到第一个镜头。

  return `第 ${r > .5 ? MT[0][1] : MT[i][1]} 天`;
});
