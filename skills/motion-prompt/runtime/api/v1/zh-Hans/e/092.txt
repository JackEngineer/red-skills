// 页面转场 / 英文名：Page transition
// 素材包：lp (落地页与网页效果) · 元素 20 · 类别：加载、导航与循环动效
// 示例内容：点击作品列表中的卡片，橙色面板从左侧扫入；URL 切换后，面板从详情页向右退场。“返回列表”也使用相同过渡。
// 动效规格：
//   - 0.8 s 点击卡片。0.9–1.3 s 面板（橙色，铺满视口，中央为标志）从左侧进入并遮住所有内容（三次方缓入缓出）。
//   - 1.3 s 在完全遮挡时，页面和地址栏 URL 同时切换（/works → /works/02）。
//   - 1.4–1.85 s 面板向右退场（三次方缓入缓出）。
//   - 3.0 s 点击“← 返回列表”：面板在 3.1–3.5 s 进入，3.5 s 切换页面，3.6–4.05 s 退场。
//   - 面板覆盖整个视口，包括页头。
// 适用场景：适合品牌网站和作品集，衔接页面切换。
// 标签：页面转场、转场、视图过渡、导航、划像转场、Barba.js 页面转场、单页应用
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#092
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 20
CELLS[19] = (k, w, h, u) => {
  frame(k, u >= 1.3 && u < 3.5 ? 'example.com/works/02' : 'example.com/works');
  const CLK = [0.8, 3.0];
  const cover = t0 => eIO(P(u, t0, t0 + 0.4)), uncover = t0 => eIO(P(u, t0, t0 + 0.45));
  // 幕布进入时左边缘从 −VW 移至 0，退出时由 0 移至 VW。
  const xs = u < 1.4 ? -VW * (1 - cover(0.9)) : u < 3.1 ? VW * uncover(1.4) : u < 3.6 ? -VW * (1 - cover(3.1)) : VW * uncover(3.6);
  const pageB = u >= 1.3 && u < 3.5;
  const [cx, cy] = path([[0, 420, 150], [0.65, 255, 100], [2.6, 255, 100], [2.85, 46, 50], [3.3, 46, 50], [4.6, 420, 150], [4.8, 420, 150]], u);
  view(k, () => {
    header(k, 0);
    if (!pageB) {
      txt(k, '作品', 34, 62, `800 18px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
      for (let i = 0; i < 3; i++) { const x = 34 + i * 152; pic(k, x, 74, 136, 60, i, 6); bars(k, x, 142, 100, 1, C.ink, 7); }
    } else {
      txt(k, '← 返回列表', 34, 55, `700 12px ${JP}`, C.sub, 'left', 'middle');
      pic(k, 34, 68, 240, 90, 1, 8);
      txt(k, '项目 02', 296, 96, `800 22px ${EN}`, C.ink); bars(k, 296, 110, 180, 3);
    }
    if (xs > -VW && xs < VW) { k.fillStyle = C.acc; k.fillRect(xs, 0, VW, VH); logo(k, xs + VW / 2 - 40, VH / 2, '#fff'); }
    click(k, cx, cy, u, CLK);
  });
};
