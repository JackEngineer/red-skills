// 吸顶页头 / 英文名：Sticky header
// 素材包：lp (落地页与网页效果) · 元素 03 · 类别：滚动效果
// 示例内容：公告条随页面滚走，下方页头则停留在顶部边缘，正文从页头下方滚过。
// 动效规格：
//   - 滚动：0.4–1.3 s 从 0→110 px，1.9–2.8 s 从 110→250 px，3.4–4.4 s 从 250→0（三次方缓入缓出）。
//   - 顶部为高 22 px 的藏青色公告条，下方是高 34 px 的页头（标志、三个菜单项、按钮）。
//   - 页头位置 = max(22 − scroll, 0)：公告条消失后，页头停在顶部边缘（position: sticky; top: 0）。
//   - 继续滚动 24 px 的过程中，页头下方 12 px 阴影的不透明度从 0 增至 0.16。
//   - 正文（照片、标题、三张卡片）正常滚动，从页头下方经过。
// 适用场景：适合长页面，让菜单和主要操作按钮始终便于使用。
// 标签：粘性定位、吸顶页头、吸顶页头、粘性定位、导航、页头、滚动
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#075
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 03
CELLS[2] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.4, 0], [1.3, 110], [1.9, 110], [2.8, 250], [3.4, 250], [4.4, 0]], u);
  const STRIP = 22;
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 70, 214, 112, 0, 8);
    txt(k, '开启全新生活，', 262, 104, `800 22px ${JP}`, C.ink); bars(k, 262, 120, 210, 3); btn(k, 262, 160, 100, 26, '了解详情');
    k.fillStyle = C.soft; k.fillRect(0, 212, VW, 240);
    txt(k, '特色', VW / 2, 246, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 262, 136, 120, 8); k.fill(); pic(k, x + 8, 270, 120, 58, i, 5); bars(k, x + 10, 340, 110, 3, C.faint, 6, 11); }
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, STRIP);
    txt(k, '限时优惠进行中', VW / 2, STRIP / 2 + 0.5, `600 11px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 250, 0.42);
  });
};
