// 组合图（柱形图＋折线图） / 英文名：Combo chart (bar + line)
// 素材包：diagram (资料中常用的图解与图表名称) · 元素 14 · 类别：展示数字的图表
// 示例内容：六根柱形按左侧刻度（灰色刻度线）向上生长，随后折线按右侧刻度（蓝色刻度线）从左到右绘出。
// 动效规格：
//   - 坐标轴：左轴及其五个刻度为灰色，右轴及其五个刻度为蓝色，带三条浅色网格线，整个循环中保留
//   - 柱形：六根宽 36 px 的浅蓝色柱形，从 0.4 s 开始，间隔 0.12 s，各用 0.36 s 向上生长（ease-out）
//   - 折线：3.5 px 蓝线在 1.3 s 至 2.2 s 以匀速从左到右连接六个点；数据点为圆环标记（半径 7 px），从 1.25 s 开始，间隔 0.18 s 弹性出现
//   - 结束：坐标轴和网格保留，仅数据在 4.15 s 至 4.55 s 淡出（单次循环 = 4.7 s）
// 适用场景：适合在同一图中展示两个单位不同的数值，例如销售额和利润率。
// 标签：组合图、组合图、柱形与折线、双坐标轴、次坐标轴、图表、数据
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 diagram -> ../../kit/diagram.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#230
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 14
const CB = [42, 55, 50, 66, 74, 88];       // 柱高以左侧刻度为准，数值为绘图区高度的百分比。
const CL = [26, 34, 52, 46, 64, 80];       // 线上的点以右侧刻度为准，数值为绘图区高度的百分比。
function combo(k, w, h, u) {
  bg(k, w, h);
  const floor = 186, top = 24, sc = (floor - top) / 100, x0 = 86, x1 = 452, slot = (x1 - x0) / CB.length, bw = 36;
  for (let i = 1; i <= 3; i++) { const y = lerp(floor, top, i / 3.4); seg(k, x0 + 4, y, x1 - 4, y, C.g2, 2); }
  seg(k, x0, top, x0, floor, C.g4, 2.5);
  seg(k, x1, top, x1, floor, C.blue, 2.5);
  seg(k, x0, floor, x1, floor, C.g4, 2.5);
  for (let i = 0; i <= 4; i++) {   // 左侧灰色刻度对应柱形，右侧蓝色刻度对应折线。
    const y = lerp(floor, top, i / 4);
    seg(k, x0 - 10, y, x0, y, C.g4, 2.5);
    seg(k, x1, y, x1 + 10, y, C.blue, 2.5);
  }
  const fo = dataFade(u);
  k.save(); k.globalAlpha = fo;
  CB.forEach((v, i) => {
    const t0 = 0.4 + i * 0.12, p = easeOut(P(u, t0, t0 + 0.36));
    if (p <= 0) return;
    const hh = v * sc * p;
    rr(k, x0 + slot * (i + 0.5) - bw / 2, floor - hh, bw, hh, 3); k.fillStyle = C.blueL; k.fill();
  });
  const pts = CL.map((v, i) => [x0 + slot * (i + 0.5), floor - v * sc]);
  const pl = P(u, 1.3, 2.2) * (pts.length - 1);
  if (pl > 0) {
    k.strokeStyle = C.blue; k.lineWidth = 3.5; k.lineJoin = 'round'; k.lineCap = 'round';
    k.beginPath(); k.moveTo(pts[0][0], pts[0][1]);
    const n = Math.floor(pl);
    for (let i = 1; i <= n; i++) k.lineTo(pts[i][0], pts[i][1]);
    if (n < pts.length - 1) { const f = pl - n; k.lineTo(lerp(pts[n][0], pts[n + 1][0], f), lerp(pts[n][1], pts[n + 1][1], f)); }
    k.stroke();
  }
  pts.forEach(([x, y], i) => {
    const s = easeOutBack(P(u, 1.25 + i * 0.18, 1.45 + i * 0.18));
    if (s > 0) { dot(k, x, y, 7 * s, C.blue); dot(k, x, y, 3 * s, C.stage); }
  });
  k.restore();
}
CELLS[13] = combo;
