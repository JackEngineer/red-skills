// 长曝光 / 英文名：Long exposure
// 素材包：photo (摄影效果、光线与风格) · 元素 11 · 类别：镜头与相机效果
// 示例内容：街道变成夜景，行驶中的汽车几乎消失，红色尾灯轨迹在车后延伸，对向车道出现白色和琥珀色光轨。
// 动效规格：
//   - 场景 2：街角，建筑、树木和路灯（x 114、292、494）位于 y 132–162 的道路后方；人物站在左前方（脚部位于 x 176、y 200，身高 124 px）。颜色变为夜景（天空 #2C3B5E，每三扇窗中的一扇为 #FFC75A）。
//   - 汽车：每 5 s 从 x −60 → 600；车身不透明度 100% → 10%。
//   - 尾灯轨迹：从 x −60 延伸到汽车位置，y 145 处为 #FF5B4D、2.6 px，y 149 处为 #FF8A6E、1.4 px。
//   - 对向车道：贯穿画面宽度的光轨分别为 y 135 处 #FFE9B8、2.6 px，y 138.5 处 #FFC75A、1.6 px，y 141.5 处 #FFE9B8、1.2 px（随效果淡入）。
//   - 右上角快门速度 1/250 → 8″。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合在一张照片中表现车流，或呈现丝滑的流水与云层。
// 标签：长曝光、光轨、慢门、夜景摄影、车灯光轨、运动
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#107
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 11
CELLS[10] = cell((k, u) => {
  const a = env(u), x = lerp(-60, 600, u / CYC);
  const p = pal(BASE, NIGHT, a);
  stageStreet(k, p, {
    car: false, person: false,
    after: D => {
      const kk = D.k;
      kk.save(); kk.globalAlpha = a; kk.lineCap = 'round';
      const x0 = -60, x1 = Math.min(x, CW + 40);
      const ln = (ya, yb, c1, c2, w1, w2) => { kk.strokeStyle = c1; kk.lineWidth = w1; kk.beginPath(); kk.moveTo(x0, ya); kk.lineTo(x1, ya); kk.stroke(); kk.strokeStyle = c2; kk.lineWidth = w2; kk.beginPath(); kk.moveTo(x0, yb); kk.lineTo(x1, yb); kk.stroke(); };
      ln(145, 149, '#FF5B4D', '#FF8A6E', 2.6, 1.4);
      for (const [y, c, lw] of [[135, '#FFE9B8', 2.6], [138.5, BASE.lit, 1.6], [141.5, '#FFE9B8', 1.2]]) { kk.strokeStyle = c; kk.lineWidth = lw; kk.beginPath(); kk.moveTo(-10, y); kk.lineTo(CW + 10, y); kk.stroke(); }
      kk.restore();
    }
  });
  k.save(); k.globalAlpha = 1 - 0.9 * a; const D = pen(k);
  car(D, x, p, { head: BASE.lit, tail: '#FF5B4D' }); k.restore();
  figure(pen(k), FIG[0], FIG[1], FIG[2], p.person);
  hud(k, a > 0.5 ? '8″' : '1/250');
});
