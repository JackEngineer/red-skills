// 鱼眼镜头 / 英文名：Fisheye lens
// 素材包：photo (摄影效果、光线与风格) · 元素 14 · 类别：镜头与相机效果
// 示例内容：街景中央向外鼓起，建筑的垂直线和道路边线向外弯曲，而画面四角保持原位。
// 动效规格：
//   - 场景 2：街角，建筑、树木和路灯（x 114、292、494）位于 y 132–162 的道路后方；人物站在左前方（脚部位于 x 176、y 200，身高 124 px）。
//   - 变形：距中心 d 的点，其距离乘以 g = (1 + k) / (1 + k·d²/R²)，R = 中心到角点的距离（289 px）。
//   - k：0 → 1.25；中心放大 2.25×，四角不动。
//   - 变形前，每条轮廓每隔 6 px 分段，使直线变成曲线。
//   - 右上角焦距 24mm → 8mm。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合大胆、活泼或窥视孔般的视角，例如滑板和音乐视频。
// 标签：鱼眼、鱼眼镜头、桶形畸变、超广角、8 毫米焦距、畸变
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#110
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 14
CELLS[13] = cell((k, u) => {
  const a = env(u), kk = 1.25 * a, cx = CW / 2, cy = CH / 2, R2 = cx * cx + cy * cy;
  const M = kk < 0.002 ? null : (x, y) => { const dx = x - cx, dy = y - cy, g = (1 + kk) / (1 + kk * (dx * dx + dy * dy) / R2); return [cx + dx * g, cy + dy * g]; };
  stageStreet(k, BASE, { M });
  hud(k, `${Math.round(lerp(24, 8, a))}mm`);
});
