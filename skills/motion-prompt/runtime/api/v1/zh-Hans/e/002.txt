// 交错动画 / 英文名：Stagger
// 素材包：motion (动态图形) · 元素 02 · 类别：入场与文字
// 示例内容：图表中的柱条从左向右依次升起，每根略微延迟。
// 动效规格：
//   - 每轮 2.35 s。七根柱条（宽 44，间距 20，高度 52, 84, 68, 112, 92, 136, 158），颜色依次循环使用蓝色、珊瑚色、黄色、薄荷绿。
//   - 第 i 根柱条（从 0 编号）在 0.15 + 0.12·i s 开始，以基线为锚点，在 0.5 s 内通过 backOut (s = 2.6) 从 0 增长至完整高度，略微超出后回稳。
//   - 白点（半径 6.5，位于柱顶上方 16 px）：比对应柱条晚 0.34 s 开始，使用弹簧曲线 1 − e^(−7t)·cos(16t)。
//   - 退场：柱条从 1.78 + 0.045·i s 开始，以三次方缓入在 0.26 s 内缩小；白点先消失，从 1.70 + 0.045·i s 开始，历时 0.2 s。
// 适用场景：适合让列表行、图表柱条、卡片等多个项目依次出现，引导视线逐项移动。
// 标签：交错动画、序列、延迟、依次入场、条形图、列表、入场
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion -> ../../kit/motion.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#002
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 02
CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();
    // 各柱顶的圆点晚于柱形出现。
    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
