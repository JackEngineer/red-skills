// 悬停抬升 / 英文名：Hover lift
// 素材包：lp (落地页与网页效果) · 元素 09 · 类别：悬停与点击交互
// 示例内容：只有光标悬停的卡片略微抬起、阴影扩散；移开后恢复原位。
// 动效规格：
//   - 三张卡片（146 × 128，圆角 10）。光标从左到右在每张卡片上停留约 1 s。
//   - 悬停时，卡片在 0.3 s 内上移 10 px（三次方缓出）；离开时在同样的 0.3 s 内返回（类似 CSS transition）。
//   - 阴影与抬升同步变化：模糊 6→22 px，y 偏移 2→12 px，不透明度 0.08→0.22。
//   - 页面浅灰，卡片白色，便于看清阴影。
// 适用场景：适合让卡片和按钮显得可以点击。
// 标签：悬停、抬升、浮起、卡片、阴影、纵向位移、交互
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#081
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 09
CELLS[8] = (k, w, h, u) => {
  frame(k);
  const CX = [26, 192, 358], CY = 18, CWD = 146, CHT = 128, LIFT = 10;
  const cur = t => path([[0, 70, 160], [0.45, 96, 78], [1.4, 110, 86], [1.62, 262, 80], [2.5, 276, 88], [2.72, 428, 80], [3.6, 440, 88], [4.0, 470, 160], [4.8, 70, 160]], t);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    for (let i = 0; i < 3; i++) {
      const inside = t => { const [x, y] = cur(t); return x > CX[i] && x < CX[i] + CWD && y > CY && y < CY + CHT; };
      const q = trans(inside, u, 0.3), x = CX[i], y = CY - LIFT * q;
      k.save(); k.shadowColor = `rgba(28,35,51,${0.08 + 0.14 * q})`; k.shadowBlur = 6 + 16 * q; k.shadowOffsetY = 2 + 10 * q;
      k.fillStyle = '#fff'; rr(k, x, y, CWD, CHT, 10); k.fill(); k.restore();
      pic(k, x + 8, y + 8, CWD - 16, 66, i, 6);
      txt(k, ['山间旅舍', '海滨旅舍', '星空旅舍'][i], x + 12, y + 96, `800 13px ${JP}`, C.ink); bars(k, x + 12, y + 106, 100, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
