// 定格 / 英文名：Freeze frame
// 素材包：camera (镜头运动与转场) · 元素 16 · 类别：拍摄方式与时间效果
// 示例内容：画面在人物跳到最高点时停住，1.4 s 后从同一帧恢复播放。
// 动效规格：
//   - 相机：完全固定，高 1.3 m；停止的是素材时间。
//   - 定格：1.6 s 停住，3.0 s 恢复，持续 1.4 s；素材时间在 0–1.6 s 正常运行，定格时保持 1.6 s，然后按 u-1.4 继续。
//   - 主体：7 m 处的人物在素材时间 0.6–2.6 s 沿抛物线跳跃，最高 1.1 m；定格恰好落在最高点，即进度 50%。
//   - 九片下落的树叶也在定格期间停止，其下落速度为 13 慢动作 中的 1.5 倍。
//   - 定格标记：5 px 蓝边，底部中央有暂停图标，68×40 px 白色圆角框内两条蓝色竖条，在 0.12 s 内淡入淡出。
//   - 右上角仅在定格期间显示“定格”，不使用交叉淡化复位。
// 适用场景：适合强调决定性瞬间，或暂停画面以叠加姓名和字幕。
// 标签：定格、静止画面、暂停、保持画面、定格画面、高亮
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera -> ../../kit/camera.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#040
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 16
CELLS[15] = shot((k, w, h, u) => {
  // 在跳跃顶点定格 1.4 秒。
  const tj = u < 1.6 ? u : u < 3.0 ? 1.6 : u - 1.4;   // 将现实经过的时间换算为画面内的时间。
  const q = P(tj, 0.6, 2.6);
  const frozen = u >= 1.6 && u < 3.0;
  world(k, w, h, cam({ y: 1.3 }), { person: { z: 7, y: jumpY(q), armUp: q > 0 && q < 1 }, extra: leaves(tj, 1.5) });
  if (frozen) {
    const a = Math.min(1, (u - 1.6) / 0.12, (3.0 - u) / 0.12);
    k.save(); k.globalAlpha = a; k.strokeStyle = C.blue; k.lineWidth = 5; k.strokeRect(2.5, 2.5, w - 5, h - 5);
    k.fillStyle = 'rgba(255,255,255,.88)'; rr(k, w / 2 - 34, h - 58, 68, 40, 8); k.fill();
    k.fillStyle = C.blue; k.fillRect(w / 2 - 12, h - 49, 8, 22); k.fillRect(w / 2 + 4, h - 49, 8, 22); k.restore();
  }
  return frozen ? '定格' : '';
});
