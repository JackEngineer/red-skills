// 示例内容：标题文字从一条线标记的不可见边界后方滑出。
// 动效规格：
//   - 每轮 2.35 s。分隔线（珊瑚色，380 × 6，y = 124）在 0.1–0.5 s 从中心向两侧展开（三次方缓入缓出），在 2.05–2.3 s 收拢消失。
//   - 上方文字（粗体 98 px，白色）：裁剪在分隔线上方（y 8–119），于 0.35–1.0 s 从下方 120 px 处向上移入，使用五次方缓出。
//   - 下方文字（32 px，黄色）：裁剪在分隔线下方（y 129–181），于 0.55–1.15 s 从上方 60 px 处向下移入，使用五次方缓出。
//   - 退场：上方文字在 1.7–2.05 s 上移 150 px（backIn, s = 1.2，先稍微下沉）；下方文字在 1.75–2.05 s 下移 70 px（三次方缓入）。
//   - 不透明度始终不变，文字仅通过离开裁剪区域而隐藏。
// 适用场景：适合让标题以干净、稳重的方式出现。
// 标签：遮罩、显现、裁切、划像转场、标题、文字、向上滑入、主标题
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion -> ../../kit/motion.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#005
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 05
CELLS[4] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, lu = mod(u, T), X = w / 2, LY = 124;
  const lg = eIO(P(lu, 0.1, 0.5)) * (1 - eIO(P(lu, 2.05, 2.3)));
  // 上行从分界线下方向上出现。
  const up = (1 - eOut5(P(lu, 0.35, 1.0))) * 120 + backIn(P(lu, 1.7, 2.05), 1.2) * -150;
  k.save(); k.beginPath(); k.rect(0, 8, w, LY - 8 - 5); k.clip();
  txt(k, '你好！', X, LY - 14 + up, `800 98px ${DISP}`, '#fff', 'center', 'alphabetic', '1px');
  k.restore();
  // 下行从分界线上方向下出现。
  const dn = (1 - eOut5(P(lu, 0.55, 1.15))) * -60 + eIn(P(lu, 1.75, 2.05)) * 70;
  k.save(); k.beginPath(); k.rect(0, LY + 5, w, 52); k.clip();
  txt(k, '初次见面', X, LY + 44 + dn, `700 32px ${JP}`, C.yel, 'center', 'alphabetic', '6px');
  k.restore();
  if (lg > 0.005) { k.fillStyle = C.coral; rr(k, X - 190 * lg, LY - 3, 380 * lg, 6, 3); k.fill(); }
};
