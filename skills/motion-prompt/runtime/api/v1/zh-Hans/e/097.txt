// 逆光 / 英文名：Backlighting
// 素材包：photo (摄影效果、光线与风格) · 元素 01 · 类别：布光与时段
// 示例内容：太阳从画面外移到人物头部正后方，人物轮廓出现一圈细亮边，浅暖色薄雾笼罩画面，朝向相机的一侧变暗。
// 动效规格：
//   - 场景 1：地平线位于 y 150，前方为头肩人像（头部中心 x 270、y 78，半径 29 px，无五官或头发）。亮部为平涂的琥珀色 #FFC75A，阴影为平涂的 #33405A。
//   - 太阳：半径 20 px，从画面外（左上方 x 40、y −40）移到头部正后方（x 270、y 56），带有半径为 2.6× 和 1.7× 的平涂光环（#FFE3A3，不透明度 35% 和 60%）。
//   - 轮廓：先用 7 px 线条描出人物轮廓，再填充人物，外侧留下 3.5 px 的亮边。
//   - 人物 #4D5A75 → #33405A；天空 #FAF9F7 → #FFF4DC；整幅画面叠加不透明度 22% 的 #FFE3A3。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合轻盈、发光的人像，或需要沐浴在光线中感觉的画面。
// 标签：逆光布光、逆光拍摄、逆光摄影、背后日光、薄雾、辉光、光晕
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#097
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 01
CELLS[0] = cell((k, u) => {
  const a = env(u);
  const BACK = Object.assign({}, BASE, { sky: '#FFF4DC', sky2: '#FFEBC4', far: '#F6E2BE', ground: '#F7EAD3', bld: '#F7E9D2', winOff: '#EEDCBF', win: '#EEDCBF', tree: '#EEDCBF', edge: '#E3CFAE', post: '#CDB894', trunk: '#E3CFAE', bulb: '#E3CFAE' });
  stagePortrait(k, pal(BASE, BACK, a), { sun: [lerp(40, PX, a), lerp(-40, PY - 22, a), 20, 1] });
  // 先用粗线描出半身像轮廓，再填充内部，使亮部只留在边缘。
  k.save(); k.globalAlpha = a; bustPath(k, PX, PY, PR); k.strokeStyle = BASE.lit; k.lineWidth = 7; k.lineJoin = 'round'; k.stroke(); k.restore();
  bust(k, PX, PY, PR, mixc(BASE.person, BASE.shade, a));
  k.save(); k.globalAlpha = 0.22 * a; k.fillStyle = BASE.glow; k.fillRect(0, 0, CW, CH); k.restore();
});
