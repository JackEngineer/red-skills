//   - 横条高 58、藏青色、与视口等宽。文字为白色 30 px、粗字重、字距 1 px。
//   - 每组四个词（网页设计、品牌设计、动态设计、界面与体验），相隔 64 px，空隙中各有橙色圆点（半径 6）。
//   - 每个循环（4.8 s）匀速向左移动一组的宽度（此处约 881 px）。先测量整组宽度，再按此宽度循环偏移，因此没有可见接缝。
//   - 重复这一组，直至填满视口宽度。
// 适用场景：适合以有活力的方式展示服务名称、客户标志或宣传语。
// 标签：跑马灯、跑马灯、无限循环、滚动文字、品牌标志条、滚动带、循环

// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#094
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 22
CELLS[21] = (k, w, h, u) => {
  frame(k);
  const WORDS = ['网页设计', '品牌设计', '动态设计', '界面与体验'], F = `800 30px ${EN}`, GAPW = 64;
  view(k, () => {
    bars(k, 34, 20, 180, 1, C.ink, 10); bars(k, 34, 38, 300, 1);
    k.fillStyle = C.ink; k.fillRect(0, 58, VW, 58);
    k.font = F; k.letterSpacing = '1px'; const ws = WORDS.map(s => k.measureText(s).width); k.letterSpacing = '0px';
    const unit = ws.reduce((a, b) => a + b + GAPW, 0);
    let x = -mod(u / CYC, 1) * unit;                 // 每轮向左滚动一组内容的宽度。
    for (let rep = 0; x < VW; rep++) WORDS.forEach((s, i) => {
      txt(k, s, x, 88, F, '#fff', 'left', 'middle', '1px');
      k.fillStyle = C.acc; k.beginPath(); k.arc(x + ws[i] + GAPW / 2, 88, 6, 0, 7); k.fill();
      x += ws[i] + GAPW;
    });
    for (let i = 0; i < 3; i++) { k.fillStyle = C.soft; rr(k, 34 + i * 152, 130, 136, 30, 6); k.fill(); bars(k, 46 + i * 152, 142, 90, 1, C.faint); }
  });
};
