// 荧光笔标记动画 / 英文名：Highlighter animation
// 素材包：lp (落地页与网页效果) · 元素 08 · 类别：滚动效果
// 示例内容：文案滚入视口时，关键词下半部会从左向右画出黄色荧光笔标记。
// 动效规格：
//   - 滚动：0.35–1.0 s 从 0→90 px，1.6–2.2 s 从 90→180 px，3.3–4.4 s 从 180→0（三次方缓入缓出）。
//   - 两行文案（20 px、粗体）分别位于页面向下 190 px 和 290 px，起初藏在视口下方。
//   - 一行的顶边经过触发线时（视口高度的 80%，134 px），其关键词下方开始画线。
//   - 笔画为黄色 #FFE36B，高 11，位于文字下半部（基线上方 8 px 至下方 3 px）。宽度在 0.8 s 内从左侧由 0 增至词宽（三次方缓出），绘制在文字后面。
//   - 从视口底部离开的文字行在不可见处重置。
// 适用场景：适合落地页文案，在阅读时突出最重要的短语。
// 标签：荧光笔、记号笔、文字高亮、强调、滚动显现、文案、下划线
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#080
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 08
CELLS[7] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.0, 90], [1.6, 90], [2.2, 180], [3.3, 180], [4.4, 0]], t), y = SY(u);
  const LINE = VH * 0.8, DUR = 0.8, MARK = '#FFE36B', F = `800 20px ${JP}`;
  const rows = [[190, ['每天的饭菜，', '只需十分钟', '即可做好。']], [290, ['即使初次下厨，', '也能轻松上手的食谱', '精选推荐。']]];
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '献给忙碌的你', 70, 50, `700 14px ${JP}`, C.sub); txt(k, '厨房里的新伙伴', 70, 80, `800 24px ${JP}`, C.ink);
    bars(k, 70, 96, 330, 4, C.faint, 6, 12, 0.8);
    for (const [base, parts] of rows) {
      k.font = F; const ws = parts.map(s => k.measureText(s).width); let x = (VW - ws.reduce((a, b) => a + b, 0)) / 2;
      const t0 = enteredAt(SY, base - 20, LINE, u), q = t0 === null ? 0 : eOut(P(u, t0, t0 + DUR));
      if (q > 0) { k.fillStyle = MARK; k.fillRect(x + ws[0], base - 8, ws[1] * q, 11); }
      parts.forEach((s, i) => { txt(k, s, x, base, F, i === 1 ? C.ink : '#4A5368'); x += ws[i]; });
      bars(k, 110, base + 22, 291, 2, C.faint, 6, 12, 0.7);
    }
    k.restore();
    sbar(k, y / 180, 0.5);
  });
};
