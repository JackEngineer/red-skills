// 聚光灯悬停 / 英文名：Spotlight hover
// 素材包：lp2 (落地页与网页效果（第 2 弹）) · 元素 10 · 类别：鼠标与输入交互
// 示例内容：光标在深色卡片上移动时，只有光标周围的圆形区域被照亮，附近卡片边缘随之发光。
// 动效规格：
//   - 视口 511 × 168：深色页面（#121828）上有三张深色卡片（147 × 116，圆角 10）。一次演示 4.7 s，每个 9.4 s 素材包周期内重复两次。
//   - 光标持续沿八字形移动：x = 255 + 190·sin(2πt/4.7)，y = 92 + 40·sin(4πt/4.7)。
//   - 仅在卡片内部，以光标为中心绘制半径 115 px 的白色径向光（中心不透明度 0.34 → 0），不溢出卡片。
//   - 卡片边框（宽 2）按距光标的距离发光，使用半径 120 px 的橙色径向光（rgb(255,122,82)）。远端边缘只保留微弱的基础线（8% 白色）。
//   - 没有滞后：光心与光标位置完全一致（通常将坐标写入 CSS 变量，作为 radial-gradient 的中心）。
// 适用场景：适合深色主题网站中的功能卡片或价格卡片行，突出指针所在区域。
// 标签：聚光灯、悬停、辉光、光标、径向渐变、深色模式、卡片
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 lp2 -> ../../kit/lp2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#154
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 10
CELLS[9] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const cx = 255 + 190 * Math.sin(2 * Math.PI * u / CYC), cy = 92 + 40 * Math.sin(4 * Math.PI * u / CYC);
  view(k, () => {
    k.fillStyle = '#121828'; k.fillRect(0, 0, VW, VH);
    txt(k, '产品特色', VW / 2, 26, `800 11px ${EN}`, 'rgba(255,255,255,.45)', 'center', 'alphabetic', '3px');
    const LAB = ['简单', '安心', '快捷'];
    for (let i = 0; i < 3; i++) {
      const x = 22 + i * 160, y = 40, cw = 147, ch = 116;
      k.fillStyle = '#1C2335'; rr(k, x, y, cw, ch, 10); k.fill();
      k.save(); rr(k, x, y, cw, ch, 10); k.clip();
      const g = k.createRadialGradient(cx, cy, 0, cx, cy, 115); g.addColorStop(0, 'rgba(255,255,255,.34)'); g.addColorStop(1, 'rgba(255,255,255,0)'); k.fillStyle = g; k.fillRect(x, y, cw, ch);
      k.restore();
      k.lineWidth = 1.5; k.strokeStyle = 'rgba(255,255,255,.08)'; rr(k, x + 0.75, y + 0.75, cw - 1.5, ch - 1.5, 10); k.stroke();
      const sg = k.createRadialGradient(cx, cy, 0, cx, cy, 120); sg.addColorStop(0, 'rgba(255,122,82,1)'); sg.addColorStop(1, 'rgba(255,122,82,0)');
      k.lineWidth = 2; k.strokeStyle = sg; rr(k, x + 1, y + 1, cw - 2, ch - 2, 10); k.stroke();
      k.strokeStyle = 'rgba(255,255,255,.75)'; k.lineWidth = 2; k.beginPath(); k.arc(x + 30, y + 32, 12, 0, 7); k.stroke();
      txt(k, LAB[i], x + 18, y + 72, `800 16px ${JP}`, 'rgba(255,255,255,.9)');
      bars(k, x + 18, y + 86, 110, 2, 'rgba(255,255,255,.16)', 5, 10);
    }
    cursor(k, cx, cy);
  });
};
