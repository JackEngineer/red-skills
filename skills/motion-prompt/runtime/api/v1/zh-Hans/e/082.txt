// 悬停图片放大 / 英文名：Image zoom on hover
// 素材包：lp (落地页与网页效果) · 元素 10 · 类别：悬停与点击交互
// 示例内容：光标悬停的照片缓慢放大，外框尺寸保持不变；移开后缩回。
// 动效规格：
//   - 两个照片框，222 × 118（圆角 8）。光标先停在左侧、再停在右侧，各约 1.5 s。
//   - 悬停时，内容以中心为基点，在 0.5 s 内从 1→1.15 放大（三次方缓出）；离开时用 0.5 s 返回。
//   - 超出外框的内容被裁切（overflow: hidden）。外框和图注不移动。
// 适用场景：适合照片网格（作品、产品、文章），提示即将被选中的项目。
// 标签：悬停、缩放、图片、比例缩放、隐藏溢出、缩略图、图库
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#082
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 10
CELLS[9] = (k, w, h, u) => {
  frame(k);
  const FX = [22, 267], FY = 16, FW = 222, FH = 118, ZOOM = 0.15;
  const cur = t => path([[0, 60, 160], [0.5, 120, 70], [1.8, 140, 84], [2.2, 370, 70], [3.4, 390, 84], [3.9, 470, 160], [4.8, 60, 160]], t);
  view(k, () => {
    for (let i = 0; i < 2; i++) {
      const inside = t => { const [x, y] = cur(t); return x > FX[i] && x < FX[i] + FW && y > FY && y < FY + FH; };
      const q = trans(inside, u, 0.5);
      pic(k, FX[i], FY, FW, FH, i, 8, 1 + ZOOM * q);
      txt(k, ['夏日山间小屋', '海边咖啡馆'][i], FX[i], FY + FH + 20, `800 13px ${JP}`, C.ink); bars(k, FX[i], FY + FH + 28, 150, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
