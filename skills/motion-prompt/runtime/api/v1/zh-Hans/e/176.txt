// 穿越镜头 / 英文名：Fly-through
// 素材包：camera2 (镜头运动与转场（第 2 弹）) · 元素 08 · 类别：镜头运动
// 示例内容：相机飞向一面带窗的墙，从窗口穿过，进入墙后的空间。
// 动效规格：
//   - 相机：向前移动 0 m → 12.5 m（高度 1.3 m，方向固定）；一面带窗的墙位于前方 8 m，相机从中穿过。
//   - 时间：0.5 s 至 3.6 s（3.1 s），三次缓入缓出（约为 cubic-bezier(0.65,0,0.35,1)）。
//   - 墙：浅灰色平面，宽 80 m、高 14 m，窗宽 2.4 m、高 2.2 m（窗台高度 0.15 m）；透过窗口可见人物（右侧 0.3 m，距离 14.5 m）、树木、建筑和山脉。
//   - 穿越：墙位于相机前方 0.25 m 时停止绘制；此时窗框已在画面之外（半宽约 1580 px），所以消失不可见。
//   - 穿过后，墙后的空间充满画面，相机靠近至距人物 2 m（#035 FPV 穿越机镜头持续在树木之间穿行）。
//   - 返回：在 3.6–4.2 s 保持，最后 0.5 s 交叉淡化回起点；右上角读数显示到窗口的距离（“距窗 8.0m” → “已穿过”）。
// 适用场景：适合以进入建筑的运动开场，或用一次运动从一个空间穿越到另一个空间。
// 标签：飞行穿越、飞行穿越、穿窗运镜、穿过、无人机、室内、运镜
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera2 -> ../../kit/camera2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#176
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 08
CELLS[7] = shot(camCell((k, w, h, p) => {
  const z = lerp(0, 12.5, p), WZ = 8;
  world(k, w, h, cam({ z, y: 1.3 }), {
    buildings: true, person: { x: 0.3, z: 14.5 },
    extra: [[0, 0, WZ, (kk, s) => {   // 用奇偶填充规则在墙面上留出窗洞。
      kk.fillStyle = C.g1; kk.beginPath(); kk.rect(-40, 0, 80, 14); kk.rect(-1.2, 0.15, 2.4, 2.2); kk.fill('evenodd');
      kk.strokeStyle = C.g3; kk.lineWidth = 1.6 / s; kk.strokeRect(-1.2, 0.15, 2.4, 2.2);
      kk.strokeStyle = C.g2; kk.lineWidth = 1.2 / s; kk.beginPath(); kk.moveTo(-40, 14); kk.lineTo(40, 14); kk.stroke();
    }]]
  });
}, p => { const d = 8 - lerp(0, 12.5, p); return d > 0 ? `距窗 ${d.toFixed(1)}m` : '已穿过'; }, 0.5, 3.6));
