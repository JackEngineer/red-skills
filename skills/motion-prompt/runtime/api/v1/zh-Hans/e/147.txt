// 堆叠卡片 / 英文名：Stacking cards
// 素材包：lp2 (落地页与网页效果（第 2 弹）) · 元素 03 · 类别：滚动效果
// 示例内容：滚动时，下一张卡片从下方升起，略微错位叠在前一张上；下层卡片略微缩小并变暗。
// 动效规格：
//   - 视口 511 × 168。滚动步进 s：0.3–1.1 s 从 0→1，1.6–2.4 s 从 1→2，3.2–4.4 s 从 2→0（三次方缓入缓出）。一次演示 4.7 s，每个 9.4 s 素材包周期内重复两次。
//   - 三张居中卡片，371 × 124、圆角 12，在页面上相隔 170 px；s 每增加一步，卡片上移 170 px。
//   - 卡片分别停在距顶部 14、26、38 px 处（position: sticky，每张的顶部偏移比前一张多 12 px），因此新卡片会盖住前一张。
//   - 每多叠一张卡片，下层卡片就以其顶边中点为中心缩小 4.5%（叠两张时缩小 9%），并叠加藏青色遮罩变暗（最高 16%）。
//   - 卡片阴影模糊 14 px，向上偏移 2 px。滚动条滑块随滚动移动。
// 适用场景：适合依次展示步骤、功能或方案，每次集中呈现一张卡片。
// 标签：堆叠卡片、粘性定位、滚动、卡片、粘性定位、步骤
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 lp2 -> ../../kit/lp2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#147
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 03
CELLS[2] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const s = steps([[0.3, 0], [1.1, 1], [1.6, 1], [2.4, 2], [3.2, 2], [4.4, 0]], u);
  const CARDS = [[C.blueL, C.ink, C.acc, '01', '策划', 0], [C.ink, '#fff', C.acc, '02', '制作', 2], [C.acc, '#fff', '#fff', '03', '交付', 1]];
  const X = 70, CWD = 371, CHT = 124, STEP = 170;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    CARDS.forEach(([bg, fg, nc, no, ttl, pv], i) => {
      const top = Math.max(14 + i * 12, 16 + i * STEP - s * STEP);
      const cov = clamp(s - i), sc = 1 - 0.045 * cov - 0.045 * clamp(s - i - 1);
      k.save(); k.translate(VW / 2, top); k.scale(sc, sc); k.translate(-VW / 2, 0);
      k.save(); k.shadowColor = 'rgba(28,35,51,.22)'; k.shadowBlur = 14; k.shadowOffsetY = -2; k.fillStyle = bg; rr(k, X, 0, CWD, CHT, 12); k.fill(); k.restore();
      txt(k, no, X + 24, 44, `800 26px ${EN}`, nc);
      txt(k, ttl, X + 24, 78, `800 20px ${JP}`, fg);
      bars(k, X + 24, 92, 140, 2, i === 0 ? '#C3CDEA' : 'rgba(255,255,255,.35)');
      pic(k, X + CWD - 160, 14, 144, CHT - 28, pv, 8);
      if (cov > 0) { k.fillStyle = `rgba(28,35,51,${0.16 * cov})`; rr(k, X, 0, CWD, CHT, 12); k.fill(); }
      k.restore();
    });
    sbar(k, s / 2, 0.4);
  });
};
