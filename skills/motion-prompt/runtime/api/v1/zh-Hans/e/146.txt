// 滚动文字显现 / 英文名：Scroll text reveal
// 素材包：lp2 (落地页与网页效果（第 2 弹）) · 元素 02 · 类别：滚动效果
// 示例内容：页面滚动时，浅灰标题从左上角开始逐字变深；向上滚动时从末尾逐字变淡。
// 动效规格：
//   - 视口 511 × 168。滚动进度 p：0.3–2.6 s 从 0→1，3.2–4.4 s 从 1→0（三次方缓入缓出）。页面上移 p × 40 px。一次演示 4.7 s，每个 9.4 s 素材包周期内重复两次。
//   - 三行标题共 24 个字，字号 27 px、字重 800、行高 38 px，起初全部为浅灰色 #D6DBE4。
//   - 第 n 个字（从 0 起计）的变深进度为 clamp(p × 26 − n)：每个字占用 1/26 的滚动进度变为正文颜色 #1C2333，与滚动同步。
//   - 反向滚动按相同公式从最后一个字开始变淡。正文条块和滚动条滑块随页面移动。
// 适用场景：适合希望读者慢慢阅读的简短宣言，例如理念或创始人寄语。
// 标签：滚动、文字显现、滚动驱动、文字排版、不透明度、叙事
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 lp2 -> ../../kit/lp2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#146
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 02
CELLS[1] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.6, 1], [3.2, 1], [4.4, 0]], u), y = p * 40;
  const LINES = ['每天的一点巧思，', '让生活一点一点', '变得更好。'], F = `800 27px ${JP}`;
  const total = LINES.join('').length;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '关于我们', 40, 34, `800 11px ${EN}`, C.acc, 'left', 'alphabetic', '2px');
    let n = 0;
    LINES.forEach((s, li) => {
      let x = 40; k.font = F;
      for (const ch of s) {
        const f = clamp(p * (total + 2) - n);
        txt(k, ch, x, 72 + li * 38, F, mix('#D6DBE4', C.ink, f));
        k.font = F; x += k.measureText(ch).width; n++;
      }
    });
    bars(k, 40, 168, 300, 3, C.faint, 6, 12, 0.7);
    k.restore();
    sbar(k, p, 0.55);
  });
};
