// 素材包：lp2 (落地页与网页效果（第 2 弹）) · 元素 17 · 类别：亮点展示
// 示例内容：点击列表照片后，同一张照片放大为详情页顶部图片，其余内容切换到详情页；返回时照片缩回原位。
// 动效规格：
//   - 视口 511 × 168，列表中三张照片为 143 × 80、圆角 6。0.75 s 点击中间照片，3.05 s 点击详情页“← 返回列表”。一次演示 4.7 s，每个 9.4 s 素材包周期内重复两次。
//   - 进入详情：0.85–1.45 s（0.6 s，三次方缓入缓出）被点击的照片持续移动并调整尺寸，从列表位置变为详情页顶部 479 × 94 图片（圆角 8）。全程始终是同一张照片。
//   - 其他照片和标题在移动的前 62% 淡出；详情标题、正文和返回按钮在后 40% 淡入。在 50% 时，URL 从 /works 变为 /works/02。
//   - 返回：3.15–3.75 s（0.6 s）以相反方向执行相同运动，回到列表位置。
//   - 这展示了列表照片与详情照片使用同一个 view-transition-name 时，View Transitions API 的效果。
// 适用场景：适合在列表和详情间往返的画廊或商品页，让用户清楚自己打开了哪一项。
// 标签：视图过渡、共享元素、页面转场、变形、图库、导航
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 lp2 -> ../../kit/lp2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#161
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 17
CELLS[16] = (k, w, h, u) => {
  u = snapU(u);
  const e = u < 2.0 ? eIO(P(u, 0.85, 1.45)) : 1 - eIO(P(u, 3.15, 3.75));
  frame(k, e > 0.5 ? 'example.com/works/02' : 'example.com/works');
  const G = [24, 184, 344], GY = 44, GW = 143, GH = 80, NM = ['雪中山间小屋', '海边咖啡馆', '夜间观景台'], PV = [0, 1, 2];
  const T = { x: 16, y: 10, w: VW - 32, h: 94 };
  const cur = t => path([[0, 420, 205], [0.5, 262, 88], [0.95, 262, 88], [1.6, 380, 150], [2.55, 380, 150], [2.9, 62, 30], [3.2, 62, 30], [3.9, 420, 205], [4.7, 420, 205]], t);
  const [cx, cy] = cur(u);
  view(k, () => {
    const ga = 1 - clamp(e * 1.6);
    k.save(); k.globalAlpha = ga;
    txt(k, '作品', 24, 30, `800 15px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    for (let i = 0; i < 3; i++) { if (i !== 1) pic(k, G[i], GY, GW, GH, PV[i], 6); txt(k, NM[i], G[i], GY + GH + 20, `800 13px ${JP}`, C.ink); bars(k, G[i], GY + GH + 30, 90, 1, C.faint, 5); }
    k.restore();
    const x = lerp(G[1], T.x, e), y = lerp(GY, T.y, e), ww = lerp(GW, T.w, e), hh = lerp(GH, T.h, e);
    pic(k, x, y, ww, hh, 1, lerp(6, 8, e));
    const da = clamp((e - 0.6) / 0.4);
    if (da > 0) {
      k.save(); k.globalAlpha = da;
      k.fillStyle = 'rgba(255,255,255,.92)'; rr(k, 26, 19, 74, 22, 11); k.fill(); txt(k, '← 返回列表', 63, 30.5, `700 11px ${JP}`, C.ink, 'center', 'middle');
      txt(k, '海边咖啡馆', 24, 134, `800 19px ${JP}`, C.ink); bars(k, 200, 122, 280, 3, C.faint, 6, 12, 0.7);
      k.restore();
    }
    click(k, cx, cy, u, [0.75, 3.05]);
  });
};
