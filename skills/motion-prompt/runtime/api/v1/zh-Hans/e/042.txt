// 匹配剪辑 / 英文名：Match cut
// 素材包：camera (镜头运动与转场) · 元素 18 · 类别：转场
// 示例内容：抛出的球在下一镜头变成同位置、同尺寸的太阳，用匹配形状连接不同场景。
// 动效规格：
//   - 衔接：2.3 s 处直接切换，0 帧；切换两侧的蓝圆位置一致，位于中心右侧 120 px、上方 46 px，半径 17 px。
//   - 前镜头（0–2.3 s）：相机固定，高 1.4 m；人物位于左侧 1.2 m、距离 7 m，从 0.5 s 开始抛球，使用三次方缓出，约 cubic-bezier(0.33,1,0.68,1)。
//   - 球：从人物头上方 2.3 m 高处移至匹配点，半径 5 px → 17 px，沿最多上拱 30 px 的弧线运动。
//   - 后镜头（2.3–4.7 s）：另一地点，没有建筑，相机高 1.2 m；相同位置的蓝圆变成太阳，在 2.4 s 内下沉 14 px。
//   - 标记：半径 26 px 的蓝色虚线圆，在切点前后各 0.7 s 内淡入淡出。
//   - 右上角显示“镜头 1” → “镜头 2”；4.7 s 时直接切回 镜头 1。
// 适用场景：适合通过共同形状或动作，连接时间或地点相距较远的两个场景。
// 标签：匹配剪辑、图形匹配剪辑、形状匹配、切镜、转场、剪辑
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera -> ../../kit/camera.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#042
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 18
CELLS[17] = shot((k, w, h, u) => {
  const cut = 2.3;
  const BX = 120, BY = -46, BR = 17;
  if (u < cut) {
    const q = eOut(P(u, 0.5, cut));
    world(k, w, h, cam({ y: 1.4 }), {
      person: { x: -1.2, z: 7, armUp: q > 0.05 }, sun: null,
      after: (kk, cc) => { const s = proj(cc, -1.2, 2.3, 7); const x = lerp(s.x, BX, q), y = lerp(s.y, BY, q) - Math.sin(q * Math.PI) * 30, r = lerp(5, BR, q); kk.fillStyle = C.blue; kk.beginPath(); kk.arc(x, y, r, 0, 7); kk.fill(); }
    });
  } else {
    const q = P(u, cut, CYC);
    world(k, w, h, cam({ y: 1.2 }), {
      buildings: false, sun: null, trees: true,
      after: (kk) => { kk.fillStyle = C.blue; kk.beginPath(); kk.arc(BX, BY + q * 14, BR, 0, 7); kk.fill(); }
    });
  }
  const g = Math.max(0, 1 - Math.abs(u - cut) / 0.7);
  if (g > 0) { k.save(); k.translate(w / 2, h / 2); k.globalAlpha = g; k.strokeStyle = C.blue; k.lineWidth = 2; k.setLineDash([5, 4]); k.beginPath(); k.arc(BX, BY + (u > cut ? P(u, cut, CYC) * 14 : 0), BR + 9, 0, 7); k.stroke(); k.restore(); }
  return u < cut ? '镜头 1' : '镜头 2';
});
