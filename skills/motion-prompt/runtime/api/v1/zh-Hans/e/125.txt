// 脉冲动画 / 英文名：Pulse
// 素材包：motion2 (动态图形（第 2 弹）) · 元素 05 · 类别：动效技巧
// 示例内容：心形像心跳一样搏动，每轮连跳两下，轮廓波纹向外扩散。
// 动效规格：
//   - 珊瑚色心形，每 1.175 s 搏动一次，每 4.7 s 四次；每次包含“咚咚”双拍。
//   - 缩放 = 1 + 0.26·G(0.2 s, width 0.065) + 0.15·G(0.44 s, width 0.07)，其中 G = e^(−((t − centre) ÷ width)²)，迅速膨胀后立即回落。
//   - 两条心形轮廓分别从 0.16 s 和 0.40 s 开始扩散，在 0.8 s 内以三次方缓出从 1.2→2.4 缩放。
//   - 扩散期间，不透明度从 0.75→0，第二条按 60% 计算；线宽从 8.5→1.5 px。
// 适用场景：适合点赞、录制或直播指示灯，以及希望用户点击的按钮，让元素更有生命力。
// 标签：脉冲动画、心跳、搏动、比例缩放、循环、点赞、吸引注意
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion2 -> ../../kit/motion2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#125
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 05
CELLS[4] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 4, lu = mod(u, T), X = w / 2, Y = h / 2 + 10;
  const g = (c, s) => Math.exp(-Math.pow((lu - c) / s, 2));
  const b = 0.26 * g(0.2, 0.065) + 0.15 * g(0.44, 0.07) + 0.26 * Math.exp(-Math.pow((lu - T - 0.2) / 0.065, 2));
  for (const [t0, s] of [[0.16, 1], [0.4, 0.6]]) {
    const q = P(lu, t0, t0 + 0.8); if (q <= 0 || q >= 1) continue;
    k.save(); k.translate(X, Y); const sc = 1.2 + 1.2 * eOut(q); k.scale(sc, sc); heartPath(k, 44);
    k.strokeStyle = C.coral; k.globalAlpha = (1 - q) * 0.75 * s; k.lineWidth = (7 * (1 - q) + 1.5) / sc; k.stroke(); k.restore();
  }
  k.globalAlpha = 1;
  k.save(); k.translate(X, Y); k.scale(1 + b, 1 + b); heartPath(k, 44); k.fillStyle = C.coral; k.fill();
  k.fillStyle = 'rgba(255,255,255,.75)'; k.beginPath(); k.ellipse(-30, -28, 9, 6, -0.7, 0, 7); k.fill(); k.restore();
};
