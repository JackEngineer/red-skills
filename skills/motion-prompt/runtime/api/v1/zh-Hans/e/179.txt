// 示例内容：透过角色的双眼观看，行走摇摆和底部的双手表明这是第一人称视角，前方人物向你招手。
// 动效规格：
//   - 相机：处于角色眼睛高度 1.6 m，向前行走 0 m → 3.2 m；0.3 s 至 4.2 s（3.9 s），三次缓入缓出（约为 cubic-bezier(0.65,0,0.35,1)）。
//   - 行走摇摆：高度 ±3 cm（每 4.7 s 6 个周期），滚转角 ±0.008 rad（3 个周期），仅在移动时叠加。
//   - 双手：底部两角为蓝色圆润的手（距中心 ±178 px），右手持白色手机；双手随身体移动，因此手保持不动，周围世界摇摆。
//   - 前方 7.5 m 的另一个人物仅在 1.6 s 至 3.4 s 之间抬起双臂挥手（看着你）；前景有两根柱子。
//   - 返回：最后 0.5 s 交叉淡化回起点（单次 = 4.7 s）；右上角读数为“主观视角”。
// 适用场景：适合让观众代入角色，或营造游戏与 VR 般的体验。
// 标签：主观视角、主观视角、第一人称视角、主观视角、眼睛、手部、拍摄角度
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera2 -> ../../kit/camera2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#179
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 11
CELLS[10] = shot(camCell((k, w, h, p, u) => {
  const z = lerp(0, 3.2, p), T = u * 2 * Math.PI / CYC;
  const bob = Math.sin(T * 6) * 0.03 * (p > 0 && p < 1 ? 1 : 0);
  world(k, w, h, cam({ z, y: 1.6 + bob, roll: Math.sin(T * 3) * 0.008 * (p > 0 && p < 1 ? 1 : 0) }), {
    person: { z: 7.5, armUp: u > 1.6 && u < 3.4, walk: 0 }, fg: [[-1.8, 2.6], [2.2, 3.8]],
    after: (kk) => {   // 双手固定在画面两侧。步行的起伏由场景的上下移动表现。
      kk.fillStyle = C.blue;
      [-1, 1].forEach(sx => { kk.save(); kk.translate(sx * 178, 64); kk.rotate(sx * -0.5); kk.fillStyle = C.blue; rr(kk, -34, -28, 68, 120, 30); kk.fill(); if (sx > 0) { kk.fillStyle = '#fff'; rr(kk, -17, -62, 34, 54, 6); kk.fill(); kk.strokeStyle = C.blue; kk.lineWidth = 3; rr(kk, -17, -62, 34, 54, 6); kk.stroke(); } kk.restore(); });
    }
  });
}, p => '主观视角', 0.3, 4.2));
