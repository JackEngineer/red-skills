// 双重曝光 / 英文名：Double exposure
// 素材包：photo (摄影效果、光线与风格) · 元素 24 · 类别：色调与后期风格
// 示例内容：背景变白，山丘、房屋、树木和太阳构成的风景缩小后，只出现在半身像的轮廓内部。
// 动效规格：
//   - 场景 3：左侧为半身像（头部中心 x 140、y 82，半径 28 px），右侧有山丘、房屋、树木和太阳（x 432、y 54，半径 15 px）。
//   - 背景：半身像外侧为白色 #FFFFFF。
//   - 半身像内部：同一片风景缩放至 0.7×，使太阳位于头部右上方，山脊横穿颈部。
//   - 内部颜色：天空 #D3DBE6，山丘 #9AA8BD 和 #76869F，地面 #55647E，树木 #3F4C66。
//   - 过渡：按效果强度将双重曝光图像叠加在原图上。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合将人物与地点、回忆或内心世界联系起来的海报和封面。
// 标签：双重曝光、多重曝光、影像叠合、剪影、超现实、图像合成
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#120
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 24
CELLS[23] = cell((k, u) => {
  const a = env(u);
  const DX = Object.assign({}, BASE, { sky: '#D3DBE6', sky2: '#D3DBE6', far: '#9AA8BD', mid: '#76869F', ground: '#55647E', bld: '#E8ECF2', edge: '#55647E', win: '#9AA8BD', winOff: '#9AA8BD', tree: '#3F4C66', trunk: '#3F4C66' });
  const draw = kk => {
    kk.fillStyle = '#FFFFFF'; kk.fillRect(0, 0, CW, CH);
    kk.save(); bustPath(kk, QX, QY, QR); kk.clip();
    // 把风景缩放至 0.7 倍，使太阳位于头部右上方、山脊位于颈部附近。
    kk.translate(QX, QY); kk.scale(0.7, 0.7); kk.translate(-415, -77);
    kk.fillStyle = DX.sky; kk.fillRect(0, -200, CW * 2, CH * 3);
    stageLand(kk, DX, { person: false });
    kk.fillStyle = DX.ground; kk.fillRect(-CW, 200, CW * 3, CH * 3);
    kk.restore();
  };
  if (a >= 0.999) { draw(k); return; }   // 效果完全显示后直接绘制，确保放大时保留清晰的轮廓。
  stageLand(k, BASE);
  if (a < 0.002) return;
  const [L, kl] = layer('dx'); draw(kl);   // 淡入和淡出期间，先在离屏画布上绘制再叠加。
  k.save(); k.globalAlpha = a; k.drawImage(L, 0, 0); k.restore();
});
