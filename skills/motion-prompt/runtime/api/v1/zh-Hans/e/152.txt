// 滚动放大 / 英文名：Zoom on scroll
// 素材包：lp2 (落地页与网页效果（第 2 弹）) · 元素 08 · 类别：滚动效果
// 示例内容：页面滚动时，中央的小照片逐渐放大并铺满窗口，周围文字淡出，照片上出现标题。
// 动效规格：
//   - 视口 511 × 168。滚动进度 p：0.3–2.3 s 从 0→1，3.0–4.4 s 从 1→0（三次方缓入缓出）。一次演示 4.7 s，每个 9.4 s 素材包周期内重复两次。
//   - 照片框从中央的 155 × 90（x = 178，y = 58，圆角 8）扩大至整个 511 × 168 窗口（圆角 0）；位置、宽度、高度和圆角均与 p 成比例变化。
//   - p 从 0→0.45 时，周围标题和正文淡出（不透明度 1 − 2.2p）。
//   - p 从 0.7→1 时，白色标题（26 px，字距 2 px）在照片中央淡入。滚动条滑块随滚动移动。
// 适用场景：适合旅行、房地产或品牌网站，让访客的注意力进入照片场景。
// 标签：缩放、滚动、图片、比例缩放、滚动驱动、首屏主视觉、全屏
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 lp2 -> ../../kit/lp2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#152
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 08
CELLS[7] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const p = steps([[0.3, 0], [2.3, 1], [3.0, 1], [4.4, 0]], u);
  const x = lerp(178, 0, p), y = lerp(58, 0, p), ww = lerp(155, VW, p), hh = lerp(90, VH, p);
  view(k, () => {
    const a = 1 - clamp(p * 2.2);
    k.save(); k.globalAlpha = a;
    txt(k, '走进风景', VW / 2, 38, `800 19px ${JP}`, C.ink, 'center');
    bars(k, 34, 66, 120, 4, C.faint, 6, 14); bars(k, 357, 66, 120, 4, C.faint, 6, 14);
    k.restore();
    pic(k, x, y, ww, hh, 0, lerp(8, 0, p));
    const b = clamp((p - 0.7) / 0.3);
    if (b > 0) { k.save(); k.globalAlpha = b; txt(k, '欢迎来到绝美风景', VW / 2, VH / 2 + 2, `800 26px ${JP}`, '#fff', 'center', 'middle', '2px'); k.restore(); }
    sbar(k, p * 0.6, 0.4);
  });
};
