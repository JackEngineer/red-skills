// 开关 / 英文名：Toggle switch
// 素材包：ui (UI 组件) · 元素 11 · 类别：切换与输入
// 示例内容：三行设置中，前两行的开关依次开启，然后再次关闭。
// 动效规格：
//   - 542×191 卡片，循环周期 4.5 s。三行设置相隔 60 px，带分隔线。x 450 处有 64×34 轨道（圆角 17）和白色滑块（半径 13，带阴影）。
//   - 点击：1.0 s 开启第 1 行，1.8 s 开启第 2 行，3.0 s 关闭第 1 行，3.6 s 关闭第 2 行。第 3 行始终关闭。
//   - 每次切换以 easeOutCubic 持续 0.2 s；滑块滑动 30 px（17 → 47）。
//   - 轨道颜色随切换进度从灰色 #d9d7d5 混合至蓝色 #1f5ae8。
// 适用场景：适合立即生效且只有两种状态的设置。
// 标签：开关切换、切换、开关控件、开关、布尔开关、设置
// 尺寸：542x191 px · 循环时长：4.5 秒
// 依赖：共用辅助代码 ui -> ../../kit/ui.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#059
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 11
CELLS[10] = ui((w, h, t) => {
  const st = [eo(seg(t, 1.0, 1.2)) * (1 - eo(seg(t, 3.0, 3.2))), eo(seg(t, 1.8, 2.0)) * (1 - eo(seg(t, 3.6, 3.8))), 0];
  for (let i = 0; i < 3; i++) {
    const y = 14 + i * 60; bar(24, y + 10, [170, 140, 190][i], C.line2, 11); bar(24, y + 30, [260, 220, 240][i], C.line, 8);
    if (i < 2) { X.fillStyle = C.line; X.fillRect(24, y + 52, 494, 1.5); }
    const k = st[i], tx = 450, ty = y + 10;
    fillRR(tx, ty, 64, 34, 17, k > .5 ? C.blue : '#d9d7d5'); if (k > 0 && k < 1) { X.globalAlpha = k; fillRR(tx, ty, 64, 34, 17, C.blue); X.globalAlpha = 1; }
    shadowed(() => circle(tx + 17 + 30 * k, ty + 17, 13, '#fff'), 4, 1, .25);
  }
  pointer([[.3, w + 30, h + 30], [.95, 486, 30], [1.35, 486, 30], [1.75, 486, 90], [2.5, 486, 90], [2.95, 486, 30], [3.25, 486, 30], [3.55, 486, 90], [3.85, 486, 90], [4.3, w + 30, h + 30]], [1.0, 1.8, 3.0, 3.6], t);
});
