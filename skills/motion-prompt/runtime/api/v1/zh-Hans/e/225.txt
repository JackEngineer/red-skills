// 素材包：diagram (资料中常用的图解与图表名称) · 元素 09 · 类别：展示数字的图表
// 示例内容：蓝色小圆点在 L 形坐标轴上从左向右弹性出现，整体向右上方散布，随后绘出虚线趋势线。
// 动效规格：
//   - 坐标轴：L 形，从 x = 96 到 448、y = 24 到 184（灰色、2.5 px），带三条浅色水平网格线，整个循环中保留
//   - 圆点：52 个蓝点（固定随机种子），半径 5 px，向右上方散布；按 x 位置在 0.4 s 至 1.65 s 之间从左到右出现，每个用 0.28 s 弹性出现（ease-out-back）
//   - 趋势线：灰色虚线（线段 7 px、间隔 8 px），在 1.85 s 至 2.5 s 从左下向右上延伸
//   - 区别于 #226 气泡图：使用大量大小相同的小点
//   - 结束：坐标轴和网格保留，仅数据在 4.15 s 至 4.55 s 淡出（单次循环 = 4.7 s）
// 适用场景：适合展示两个数值是否相关，例如一个数值是否随另一个上升。
// 标签：散点图、散点图、相关性、圆点、趋势线、直角坐标图、数据
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 diagram -> ../../kit/diagram.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#225
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 09
const SCPTS = (() => {
  const r = rand(907), pts = [];
  for (let i = 0; i < 52; i++) {
    const x = 0.04 + r() * 0.92;
    const y = clamp(0.1 + 0.74 * x + (r() + r() - 1) * 0.2, 0.05, 0.95);
    pts.push({ x, y });
  }
  return pts.sort((a, b) => a.x - b.x);
})();
function scatter(k, w, h, u) {
  bg(k, w, h);
  const A = SC; axes(k, A.x0, A.y0, A.x1, A.y1);
  const fo = dataFade(u);
  k.save(); k.globalAlpha = fo;
  const X = x => lerp(A.x0 + 10, A.x1 - 8, x), Y = y => lerp(A.y1 - 8, A.y0 + 6, y);
  // 趋势线在散点之后绘制。
  const tl = easeInOut(P(u, 1.85, 2.5));
  if (tl > 0) {
    const xa = X(0.02), ya = Y(0.11), xb = X(0.98), yb = Y(0.83);
    seg(k, xa, ya, lerp(xa, xb, tl), lerp(ya, yb, tl), C.g4, 2.5, [7, 8]);
  }
  for (const p of SCPTS) {
    const t0 = 0.4 + p.x * 1.25;
    const s = easeOutBack(P(u, t0, t0 + 0.28));
    dot(k, X(p.x), Y(p.y), 5 * s, C.blue);
  }
  k.restore();
}
CELLS[8] = scatter;
