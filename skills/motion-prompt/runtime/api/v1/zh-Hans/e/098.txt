// 轮廓光 / 英文名：Rim light
// 素材包：photo (摄影效果、光线与风格) · 元素 02 · 类别：布光与时段
// 示例内容：周围环境变暗，人物左后方和右后方的两盏灯只在头肩两侧勾出细亮边；右下方的俯视图标出两盏灯的位置。
// 动效规格：
//   - 场景 1：地平线位于 y 150，前方为头肩人像（头部中心 x 270、y 78，半径 29 px，无五官或头发）。
//   - 背景：明亮场景 → 暗场景（天空 #2B2D33、地面 #24252A）；人物 #4D5A75 → #1F2128。
//   - 亮边：将两个琥珀色 #FFC75A 的人物副本分别向左、右偏移 3.5 px，并向上偏移 2.5 px，放在深色人物后方（形成 3.5 px 宽的亮边）。
//   - 右下方俯视图（66×66 px）：灯光位于 ±135°（左后方和右后方）；右上角读数为“±135°”。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合让人物或产品从暗背景中分离出来，突出外形。
// 标签：轮廓光、轮廓布光、轮廓光、逆光、侧后轮廓光、轮廓、影棚
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#098
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 02
CELLS[1] = cell((k, u) => {
  const a = env(u);
  stagePortrait(k, pal(BASE, DARK, a));
  k.save(); k.globalAlpha = a; k.fillStyle = BASE.lit;
  for (const dx of [-3.5, 3.5]) { k.save(); k.translate(dx, -2.5); bustPath(k, PX, PY, PR); k.fill(); k.restore(); }
  k.restore();
  bust(k, PX, PY, PR, mixc(BASE.person, DARK.person, a));
  plan(k, [[Math.PI * 0.75, a], [-Math.PI * 0.75, a]]);
  hud(k, '±135°');
});
