// 中继器 / 英文名：Repeater
// 素材包：motion2 (动态图形（第 2 弹）) · 元素 14 · 类别：文字与图形
// 示例内容：一个形状沿圆周复制，各副本略微错开节奏地旋转和脉动。
// 动效规格：
//   - 圆角条（24 × 60，圆角半径 12）最多复制 12 个，沿画面中心半径 58 的圆排列，每个朝外旋转，角度加 90°。
//   - 数量在 0.35–1.7 s 从 1→12，在 3.6–4.45 s 从 12→1，使用二次方缓入缓出。每个新副本以三次方缓出从前一个位置滑到自己的位置，向前 30°。
//   - 整个圆环每 4.7 s 转一圈。每个副本每 4.7 s 以 1 ± 0.22 脉动两次，相邻副本相位滞后 0.55 rad，形成环形波纹。
//   - 颜色依次为蓝色、珊瑚色、黄色、薄荷绿。原始副本有白色描边，右上角显示“×N”计数。
// 适用场景：适合加载动画、背景纹样、花朵或光芒等由同一形状重复构成的图案。
// 标签：中继器、复制、径向、阵列、复制图形、图案、加载指示器
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion2 -> ../../kit/motion2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#134
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 14
CELLS[13] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const N = 12, X = w / 2, Y = h / 2, R = 58, spin = 2 * Math.PI * u / CYC;
  const cnt = 1 + (N - 1) * (eIOs(P(u, 0.35, 1.7)) - eIOs(P(u, 3.6, 4.45)));
  for (let i = N - 1; i >= 0; i--) {
    const vis = i === 0 ? 1 : clamp(cnt - i); if (vis <= 0) continue;
    const ang = spin + (i === 0 ? 0 : (i - 1 + eOut(vis))) * 2 * Math.PI / N;
    const pul = (1 + 0.22 * Math.sin(2 * Math.PI * 2 * u / CYC - i * 0.55)) * (0.55 + 0.45 * vis);
    k.save(); k.translate(X + Math.cos(ang) * R, Y + Math.sin(ang) * R); k.rotate(ang + Math.PI / 2); k.scale(pul, pul);
    k.fillStyle = PAL[i % 4]; rr(k, -12, -30, 24, 60, 12); k.fill();
    if (i === 0) { k.strokeStyle = '#fff'; k.lineWidth = 4; rr(k, -12, -30, 24, 60, 12); k.stroke(); }
    k.restore();
  }
  txt(k, '×' + Math.round(cnt), w - 34, 46, `800 40px ${DISP}`, C.yel, 'right', 'middle');
};
