// 突切 / 英文名：Smash cut
// 素材包：camera (镜头运动与转场) · 元素 22 · 类别：转场
// 示例内容：人物熟睡的安静场景毫无预兆地直切到强烈场景，音量表同时猛增。
// 动效规格：
//   - 衔接：2.4 s 处直接切换，0 帧，无转场效果，依靠反差产生冲击。
//   - 前镜头（0–2.4 s）：相机固定，高 1.2 m，焦距 330 px；人物躺在 7 m 处，三个“呼”字上浮并淡出；8 格音量表约亮 0–1 格，值 0.08±0.04。
//   - 后镜头（2.4–4.2 s）：焦距变为 2.6 倍，即 858 px，形成紧构图；滚转 -0.12 rad，约 -7°；人物举起双臂，周围有 18 根蓝色放射线。
//   - 后镜头震动：滚转 ±0.03 rad，每秒约 6 次；偏航 ±0.006 rad，每秒约 8 次；音量表为 0.85±0.15，接近满格。
//   - 返回：最后 0.5 s 交叉淡化回安静镜头，二次方缓入缓出。
//   - 右上角显示“♪ 安静”→“♪ 高音量”。
// 适用场景：适合喜剧或惊吓，在瞬间展现平静与混乱、预期与现实的反差。
// 标签：突切、硬切、突变、反差、由静转响、喜剧、转场
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera -> ../../kit/camera.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#046
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 22
function meter(k, w, h, lv) {
  const x = 24, y = h - 22, n = 8;
  for (let i = 0; i < n; i++) { k.fillStyle = (i + 0.5) / n < lv ? C.blue : C.g2; k.fillRect(x + i * 9, y - 6 - i * 2.2, 6, 6 + i * 2.2); }
}
CELLS[21] = shot((k, w, h, u) => {
  const cut = 2.4, r = P(u, CYC - RESET, CYC);
  const quiet = (kk, uu) => {
    world(kk, w, h, cam({ y: 1.2 }), {
      buildings: false, sun: [24, 26, 170], sunCol: C.blueXL, extra: [[0.4, 0, 7, (k2) => { k2.fillStyle = C.blue; rr(k2, -0.8, 0, 1.6, 0.34, 0.15); k2.fill(); k2.beginPath(); k2.arc(-0.95, 0.24, 0.17, 0, 7); k2.fill(); }]],
      after: (k2, cc) => { const p0 = proj(cc, -0.6, 1.0, 7); for (let i = 0; i < 3; i++) { const ph = mod(uu * 0.35 + i / 3, 1); k2.globalAlpha = Math.sin(ph * Math.PI) * 0.8; txt(k2, '呼', p0.x - 10 - ph * 20 - i * 4, p0.y - ph * 50, `600 ${16 + i * 4}px ${EN}`, C.sub, 'center', 'middle'); } k2.globalAlpha = 1; }
    });
    meter(kk, w, h, 0.08 + 0.04 * Math.sin(uu * 5));
  };
  const loud = (kk, uu) => {
    const T = uu * 40;
    const c = cam({ y: 1.25, f: F0 * 2.6, roll: -0.12 + Math.sin(T) * 0.03, yaw: Math.sin(T * 1.3) * 0.006 });
    world(kk, w, h, c, {
      buildings: false, trees: false, mountains: false, grid: true, sun: null,
      person: { x: 0, z: 7, armUp: true, walk: 0.6 },
      after: (k2) => { k2.strokeStyle = C.blue; k2.lineWidth = 3.5; for (let i = 0; i < 18; i++) { const a = i / 18 * Math.PI * 2 + 0.1; const r0 = 92 + (i % 3) * 14; k2.beginPath(); k2.moveTo(Math.cos(a) * r0 * 1.7, Math.sin(a) * r0); k2.lineTo(Math.cos(a) * 500, Math.sin(a) * 300); k2.stroke(); } }
    });
    meter(kk, w, h, 0.85 + 0.15 * Math.sin(uu * 31));
  };
  if (u < cut) quiet(k, u);
  else blend(k, kk => loud(kk, u), kk => quiet(kk, 0), eIOs(r));
  return u < cut || r > .5 ? '♪ 安静' : '♪ 高音量';
});
