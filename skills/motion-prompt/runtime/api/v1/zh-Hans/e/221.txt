// 思维导图 / 英文名：Mind map
// 素材包：diagram (资料中常用的图解与图表名称) · 元素 05 · 类别：整理思路的图
// 示例内容：六条曲线分支从中央的“主题”向两侧生长，随后长出更细的枝条和小文字节点。
// 动效规格：
//   - 中心：100×44 px 的蓝色圆角框，白色文字“主题”（19 px），从 0.4 s 开始用 0.3 s 弹性出现（ease-out-back）
//   - 分支：六条三次贝塞尔曲线（3 px），左右各三条，从 0.7 s 开始，间隔 0.09 s，各用 0.5 s 生长，末端接 54×24 px 的节点；中蓝色和灰色交替
//   - 细枝：从 1.45 s 开始，分支末端用 0.35 s 长出更细的 2.5 px 曲线，每条末端接 40×18 px 的小节点
//   - 形态：曲线分支与文字节点，不同于 #240 同心圆图的嵌套圆环
//   - 结束：整个图形在 4.1 s 至 4.5 s 淡出，卡片保持空白，直到 4.7 s 循环重新开始
// 适用场景：适合头脑风暴，或围绕一个主题展开并整理想法。
// 标签：思维导图、头脑风暴、创意、径向、分支、中心主题、思维导图
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 diagram -> ../../kit/diagram.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#221
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 05
const MM = [
  // 每项为 [分支侧, 终点横向偏移, 终点纵向偏移, 颜色, 子分支列表 [横向偏移, 纵向偏移]]。
  [1, 128, -52, C.blueM, [[204, -72], [204, -32]]],
  [1, 142, 8, C.g4, []],
  [1, 124, 56, C.blueM, [[204, 76]]],
  [-1, 128, -52, C.g4, [[204, -70]]],
  [-1, 140, 10, C.blueM, []],
  [-1, 124, 56, C.g4, [[204, 36], [204, 76]]],
];
function mindmap(k, w, h, u) {
  bg(k, w, h);
  k.globalAlpha = fade(u);
  const cx = w / 2, cy = h / 2;
  MM.forEach(([sd, ex, ey, col, twigs], i) => {
    const x0 = cx + sd * 40, x1 = cx + sd * ex, y1 = cy + ey;
    const pts = bezPts([x0, cy], [cx + sd * 78, cy], [cx + sd * (ex - 50), y1], [x1 - sd * 26, y1]);
    pen(k, col, 3); trace(k, pts, easeInOut(P(u, 0.7 + i * 0.09, 1.2 + i * 0.09)));
    twigs.forEach(([tx, ty], j) => {
      const a = [x1 + sd * 26, y1], b = [cx + sd * tx - sd * 20, cy + ty], t0 = 1.45 + i * 0.07 + j * 0.05;
      const tp = bezPts(a, [a[0] + sd * 22, a[1]], [b[0] - sd * 26, b[1]], b, 16);
      pen(k, col, 2.5); trace(k, tp, easeInOut(P(u, t0, t0 + 0.35)));
      box(k, cx + sd * tx, cy + ty, 40, 18, 9, pop(u, t0 + 0.3, 0.25), col === C.g4 ? C.g3 : C.blueL);
    });
    box(k, x1, y1, 54, 24, 12, pop(u, 1.15 + i * 0.09, 0.28), col === C.g4 ? C.g4 : C.blueM);
  });
  const s = pop(u, 0.4);
  if (s > 0.01) {
    k.save(); k.translate(cx, cy); k.scale(s, s);
    k.fillStyle = C.blue; rr(k, -50, -22, 100, 44, 22); k.fill();
    txt(k, '主题', 0, 1, `700 19px ${FONT}`, '#FFFFFF', 'center', 'middle');
    k.restore();
  }
  k.globalAlpha = 1;
}
CELLS[4] = mindmap;
