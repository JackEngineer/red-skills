// 分割光 / 英文名：Split lighting
// 素材包：photo (摄影效果、光线与风格) · 元素 04 · 类别：布光与时段
// 示例内容：光线从接近正面（20°）转到正侧面（90°）；头部阴影从月牙形扩大到恰好一半，将脸从中间分成明暗两侧。
// 动效规格：
//   - 场景 1：地平线位于 y 150，前方为头肩人像（头部中心 x 270、y 78，半径 29 px，无五官或头发）。背景变暗（天空 #E7E5E1）。
//   - 灯光：从画面左侧照入，角度 20° → 90°（右下方为俯视图，右上角读数“90° · ↑0°”）。
//   - 头部：x = 270 + cos(angle)·√(29² − dy²) 右侧为阴影 #33405A，左侧为亮部 #FFC75A；角度为 90° 时，边界是一条垂直中线。
//   - 身体（半宽 56 px）：x = 270 + cos(angle)·56 右侧为阴影，左边缘的一条窄带受光。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合戏剧化、紧张或情绪感较强的人像。
// 标签：分割光、侧光、人像布光、半边脸、戏剧感、明暗对照法
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#100
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 04
CELLS[3] = cell((k, u) => {
  const a = env(u), az = lerp(0.35, Math.PI / 2, a);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a), W = torsoW(PR);
  bust(k, PX, PY, PR, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, PR, 0, TAU); kk.fill();
    headShade(kk, PX, PY, PR, az, 1, sh);
    torso(kk, PX, PY, PR, PX - W, PX - Math.cos(az) * 0.4 * W - 0.4 * W, lit);
    torso(kk, PX, PY, PR, PX + Math.cos(az) * W, PX + W, sh);
  });
  plan(k, [[az, a]]);
  hud(k, `${Math.round(az * 180 / Math.PI)}° · ↑0°`);
});
