// 动效规格：
//   - 全屏海景在 0.45–0.95 s 分成两栏，分隔线从 539→269.5；在 1.4–1.9 s 分成三栏，分隔线位于 1/3 和 2/3；在 3.75–4.4 s 合回一栏，均使用三次方缓入缓出。
//   - 三个场景（海、城市、田野）分别在自己的面板内居中、裁剪并持续运动：海浪和帆船，城市窗户每 4.7 s 切换 8 次，风车每 4.7 s 转两圈。
//   - 面板之间为 6 宽的白色分隔线。
// 适用场景：适合比较、展示同时发生的事，或并列多个人物和地点，例如前后对比或多个选项。
// 标签：分屏、多屏、面板、并排、对比、布局
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion2 -> ../../kit/motion2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#144
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 24
CELLS[23] = (k, w, h, u) => {
  u = snapU(u);
  const s2 = eIO(P(u, 0.45, 0.95)), s3 = eIO(P(u, 1.4, 1.9)), m = eIO(P(u, 3.75, 4.4));
  let b1 = lerp(lerp(w, w / 2, s2), w / 3, s3), b2 = lerp(w, 2 * w / 3, s3);
  b1 = lerp(b1, w, m); b2 = lerp(b2, w, m);
  pnSea(k, 0, b1, h, u, false);
  if (b1 < w - 0.5) pnCity(k, b1, Math.max(b1, b2), h, u, false);
  if (b2 < w - 0.5) pnField(k, b2, w, h, u, false);
  k.fillStyle = '#fff';[b1, b2].forEach(b => { if (b < w - 0.5) k.fillRect(b - 3, 0, 6, h); });
};
