// 预加载画面 / 英文名：Preloader
// 素材包：lp (落地页与网页效果) · 元素 17 · 类别：加载、导航与循环动效
// 示例内容：藏青色屏幕上，标志下方数字从 0% 增至 100%；随后整屏上移退场，下方页面淡入。
// 动效规格：
//   - 藏青色屏幕铺满视口，包含标志、数字（28 px）和进度条（180 × 4）。
//   - 0.25–2.1 s 从 0→100%（三次方缓出，取整）；进度条同步填充橙色。
//   - 2.3–2.85 s 整屏上移 168 px 并退场（三次方缓入缓出）。
//   - 页面标题和照片在 2.35–2.95 s 从 0→1 淡入，并从下方 14 px 升起（三次方缓出）。
//   - 4.25–4.75 s 加载屏从顶部重新降下，数值为 0%（下一次加载）。
// 适用场景：适合图片或视频较多的网站，让加载等待也能展示品牌。
// 标签：预加载画面、加载画面、加载指示器、百分比、开场、启动画面、帘幕
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#089
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 17
CELLS[16] = (k, w, h, u) => {
  frame(k);
  const n = Math.round(100 * eOut(P(u, 0.25, 2.1)));
  const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
  const show = eOut(P(u, 2.35, 2.95));
  view(k, () => {
    header(k, 0);
    k.save(); k.globalAlpha = show; k.translate(0, 14 * (1 - show));
    txt(k, '欢迎光临。', 34, 82, `800 26px ${JP}`, C.ink); bars(k, 34, 98, 200, 3); btn(k, 34, 136, 96, 24, '开始使用');
    pic(k, 270, 46, 214, 112, 0, 8); k.restore();
    if (cy > -VH) {
      k.save(); k.translate(0, cy); k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
      logo(k, VW / 2 - 40, 56, '#fff');
      txt(k, (u >= 4.25 ? 0 : n) + '%', VW / 2, 98, `800 28px ${EN}`, '#fff', 'center', 'middle');
      k.fillStyle = 'rgba(255,255,255,.15)'; rr(k, VW / 2 - 90, 122, 180, 4, 2); k.fill();
      k.fillStyle = C.acc; rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2); k.fill();
      k.restore();
    }
  });
};
