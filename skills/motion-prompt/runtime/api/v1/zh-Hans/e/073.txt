// 视差滚动 / 英文名：Parallax scrolling
// 素材包：lp (落地页与网页效果) · 元素 01 · 类别：滚动效果
// 示例内容：页面向下滚动时，标题正常上移，天空与太阳、远山和近山则以较慢速度移动。
// 动效规格：
//   - 浏览器窗口的视口为 511 × 168。滚动：0.3–1.9 s 从 0→120 px，2.9–4.3 s 从 120→0（三次方缓入缓出）。
//   - 各图层的位移为滚动量 × 系数：天空与太阳 0.15，远山 0.4，近山 0.7。
//   - 标题（28 px）、按钮和下一区块按 1.0 移动（正常滚动）。
//   - 下一区块为白色，从首屏底部（168 px）开始，滑过移动较慢的背景。
//   - 右侧滚动条滑块随滚动移动。
// 适用场景：适合为首屏或区块照片增加纵深感。
// 标签：视差、视差滚动、滚动、纵深、图层、首屏主视觉、背景
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#073
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 01
CELLS[0] = (k, w, h, u) => {
  frame(k);
  const s = steps([[0.3, 0], [1.9, 1], [2.9, 1], [4.3, 0]], u), y = s * 120;
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 200); g.addColorStop(0, '#9DB6FB'); g.addColorStop(1, '#E6ECFF');
    k.save(); k.translate(0, -y * 0.15);                         // 天空和太阳以滚动速度的 0.15 倍移动。
    k.fillStyle = g; k.fillRect(0, 0, VW, 260); k.fillStyle = C.acc; k.beginPath(); k.arc(392, 54, 25, 0, 7); k.fill(); k.restore();
    k.save(); k.translate(0, -y * 0.4); mount(k, VW, 190, M1, '#7D9AF4', 0); k.restore();   // 远山以滚动速度的 0.4 倍移动。
    k.save(); k.translate(0, -y * 0.7); mount(k, VW, 205, M2, '#4B6BDB', 1); k.restore();   // 近山以滚动速度的 0.7 倍移动。
    k.save(); k.translate(0, -y);                                // 标题、按钮和下一节按正常滚动速度移动。
    txt(k, '向山野出发。', 34, 64, `800 28px ${JP}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 34, 78, 160, 2, 'rgba(255,255,255,.85)');
    btn(k, 34, 106, 92, 26, '立即预订');
    k.fillStyle = '#fff'; k.fillRect(0, VH, VW, 200);
    txt(k, '热门路线', 34, VH + 34, `800 16px ${JP}`, C.ink);
    for (let i = 0; i < 3; i++) { pic(k, 34 + i * 152, VH + 50, 136, 58, i, 6); bars(k, 34 + i * 152, VH + 116, 110, 1, C.faint); }
    k.restore();
    sbar(k, s * 0.35, 0.4);
  });
};
