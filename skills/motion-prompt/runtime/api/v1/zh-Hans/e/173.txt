// 素材包：camera2 (镜头运动与转场（第 2 弹）) · 元素 05 · 类别：镜头运动
// 示例内容：相机平滑向前跟随行走的人物，仅有缓慢轻柔的摇摆，没有手持拍摄般的抖动。
// 动效规格：
//   - 相机：以恒定 1.7 m/s 向前移动（每 4.7 s 移动 8 m）；高度 1.45 m，俯仰角 -0.02 rad；无缓动。
//   - 摇摆：高度 ±3.5 cm（每 4.7 s 2 个周期）、横向 ±5 cm（1 个周期）、偏航角 ±0.012 rad、滚转角 ±0.006 rad，均为整数周期的正弦波，缓慢且幅度宽（#034 手持镜头快速且不规则）。
//   - 主体：人物位于相机前方 5 m，以相同速度行走；腿和手臂每 4.7 s 完成 5 次步幅。
//   - 树木：x = -5 m 处高 3.6 m，x = +5 m 处高 3.2 m，每隔 8 m 一棵（两排错开 4 m）；山脉和太阳跟随相机移动。
//   - 8 m 是网格（2 m）和树木（8 m）间距的倍数，因此每 4.7 s 无缝循环；右上角读数为“斯坦尼康”。
// 适用场景：适合让步行跟拍显得平稳且具有电影感。
// 标签：斯坦尼康、云台、稳定器、平稳运动、滑行、悬浮、运镜
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera2 -> ../../kit/camera2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#173
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 05
CELLS[4] = shot((k, w, h, u) => {
  const v = 8 / CYC, z = u * v, T = u * 2 * Math.PI / CYC;   // 相机每轮前进 8 米，恰好跨过四格地面网格和一段树距，首尾画面可以对齐。
  const c = cam({ z, y: 1.45 + 0.035 * Math.sin(T * 2), x: 0.05 * Math.sin(T), yaw: 0.012 * Math.sin(T + 1), roll: 0.006 * Math.sin(T * 2 + 0.5), pitch: -0.02 });
  const ex = [], L = 8, base = Math.floor(z / L) * L;
  for (let i = 0; i < 10; i++) { const zz = base + i * L; ex.push([-5, 0, zz + 2, (kk, s) => tree(kk, s, 3.6)]); ex.push([5, 0, zz + 6, (kk, s) => tree(kk, s, 3.2, C.blueXL)]); }
  world(k, w, h, c, { buildings: false, trees: false, extra: ex, mtnZ: z, sun: [30, 30, z + 170], gridX: 20, person: { z: z + 5, walk: T * 5 } });
  return '斯坦尼康';
});
