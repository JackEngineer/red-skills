// 可变字体动画 / 英文名：Variable font animation
// 素材包：motion2 (动态图形（第 2 弹）) · 元素 12 · 类别：文字与图形
// 示例内容：同一个词在纤细与极粗字重之间平滑往返，下方滑块显示当前字重。
// 动效规格：
//   - “动态设计”（112 px，字距 2 px）。字重每 4.7 s 往返两次，v = 0.5 − 0.5·cos(2π × 2 × 时间 ÷ 4.7)，两端带缓动。
//   - 字重 = round(100 + 800v)，从 100 到 900，直接设置为字体字重。
//   - 下方滑杆从 x = 112→427，y = 180，左侧标细、右侧标粗。黄色滑块（半径 14）随 v 移动，上方显示当前字重。
// 适用场景：适合标题、标志或单个词，通过字形本身表达音量或情绪。
// 标签：可变字体、字重、文字排版、文字、字重、动态文字
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion2 -> ../../kit/motion2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#132
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 12
CELLS[11] = (k, w, h, u) => {
  stage(k);
  const v = 0.5 - 0.5 * Math.cos(2 * Math.PI * 2 * u / CYC), wt = Math.round(100 + 800 * v);
  txt(k, '动态设计', w / 2, 80, `${wt} 112px ${JP}`, '#fff', 'center', 'middle', '2px');
  const x0 = 112, x1 = w - 112, y = 180, kx = x0 + (x1 - x0) * v;
  txt(k, '细', x0 - 34, y + 1, `300 30px ${JP}`, C.dim, 'center', 'middle');
  txt(k, '粗', x1 + 34, y + 1, `900 30px ${JP}`, C.dim, 'center', 'middle');
  k.fillStyle = C.navy3; rr(k, x0, y - 5, x1 - x0, 10, 5); k.fill();
  k.fillStyle = C.blue; rr(k, x0, y - 5, Math.max(10, kx - x0), 10, 5); k.fill();
  k.fillStyle = C.yel; k.beginPath(); k.arc(kx, y, 14, 0, 7); k.fill();
  txt(k, String(wt), kx, y - 30, `800 24px ${DISP}`, C.yel, 'center', 'middle');
};
