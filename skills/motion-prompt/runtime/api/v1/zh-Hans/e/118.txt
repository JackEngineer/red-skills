// 胶片颗粒 / 英文名：Film grain
// 素材包：photo (摄影效果、光线与风格) · 元素 22 · 类别：色调与后期风格
// 示例内容：同一片风景的整个画面上出现细颗粒，颜色略微偏向温暖的奶油色。
// 动效规格：
//   - 场景 3：左侧为半身像（头部中心 x 140、y 82，半径 28 px），右侧有山丘、房屋、树木和太阳（x 432、y 54，半径 15 px）。
//   - 颗粒：2,600 个方块（固定随机种子），大小 0.8–1.7 px；55% 为深色 #3A3530，45% 为白色。
//   - 颗粒不透明度：16–38% × 效果强度；颗粒不移动（没有闪烁）。
//   - 颜色：所有颜色向 #F2DDBE 偏移 16%。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合减少干净的数码感，增添模拟胶片质感。
// 标签：胶片颗粒、颗粒、噪声、胶片质感、35 毫米胶片、纹理、暖色调
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#118
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 22
CELLS[21] = cell((k, u) => {
  const a = env(u);
  const FILM = mapPal(BASE, c => mixc(c, '#F2DDBE', 0.16));
  const p = pal(BASE, FILM, a);
  const G = once('grain', () => { const r = rng(22), o = []; for (let i = 0; i < 2600; i++) o.push([r() * CW, r() * CH, 0.8 + r() * 0.9, r() < 0.55 ? 0 : 1, 0.16 + r() * 0.22]); return o; });
  const grain = kk => { kk.save(); for (const [x, y, s, d, al] of G) { kk.globalAlpha = al * a; kk.fillStyle = d ? '#FFFFFF' : '#3A3530'; kk.fillRect(x, y, s, s); } kk.restore(); };
  stageLand(k, p); grain(k);
});
