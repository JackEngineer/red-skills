// 黑白 / 英文名：Black and white
// 素材包：photo (摄影效果、光线与风格) · 元素 19 · 类别：色调与后期风格
// 示例内容：同一片风景的颜色逐渐褪去，太阳和人物受光处变成灰色，只靠明暗区分形状。
// 动效规格：
//   - 场景 3：左侧为半身像（头部中心 x 140、y 82，半径 28 px），右侧有山丘、房屋、树木和太阳（x 432、y 54，半径 15 px）。
//   - 转换：各颜色变为对应亮度的灰色（0.2126R + 0.7152G + 0.0722B），以 140 为中心将对比度乘以 ×1.25。
//   - 例如人物 #4D5A75 → 约 #4D4D4D，太阳 #F5A524 → 约 #B5B5B5。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合营造历久弥新的感觉，或强调形状、光线和表情而非颜色。
// 标签：黑白、单色、黑白、灰度、去饱和、经典
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#115
// 以上资源路径相对于本规格文件。

// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 19
CELLS[18] = cell((k, u) => {
  const a = env(u);
  const GRAY = mapPal(BASE, c => { const g = clamp((lum(c) - 140) * 1.25 + 140, 0, 255); return toHex([g, g, g]); });
  stageLand(k, pal(BASE, GRAY, a));
});
