// 交叉剪辑 / 英文名：Cross-cutting
// 素材包：camera2 (镜头运动与转场（第 2 弹）) · 元素 22 · 类别：转场与剪辑
// 示例内容：跑步者和汽车两个同时发生的场景交替出现，剪切间隔越来越短。
// 动效规格：
//   - 两个场景共用同一时钟：A 是跑步者（左侧 0.3 m，纵深 15 m → 4.5 m，腿部按 u×12 运动），B 是城市中的汽车（右侧 1 m，纵深 17 m → 4.5 m）；两者均在 4.7 s 内靠近。
//   - 剪切：在 0 / 1.1 / 2.0 / 2.7 / 3.3 / 3.8 / 4.2 / 4.5 s 处在 A 和 B 之间硬切；镜头时长依次缩短为 1.1 → 0.9 → 0.7 → 0.6 → 0.5 → 0.4 → 0.3 s，以增加紧张感。
//   - 另一场景显示时，每个场景仍继续推进（切回来时主体已移动）。
//   - 相机：A 高度 1.3 m，固定，前景有四根柱子；B 高度 1.6 m，俯仰角 +0.06 rad，固定，场景含城市和太阳。
//   - 底部条带：A（灰色）和 B（蓝色）交替的八段，带播放头；最后 0.3 s 交叉淡化回 A 的起点。
//   - 右上角读数为“场景 A” / “场景 B”。
// 适用场景：适合在同时发生的事件之间制造紧张感，例如追逐或救援。
// 标签：交叉剪辑、平行剪辑、交叉剪辑、并行动作、追逐、剪辑
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera2 -> ../../kit/camera2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#190
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 22
const XC = [0, 1.1, 2.0, 2.7, 3.3, 3.8, 4.2, 4.5, CYC];
CELLS[21] = shot((k, w, h, u) => {
  const r = P(u, CYC - 0.3, CYC);
  const runA = (kk, t) => world(kk, w, h, cam({ y: 1.3 }), { person: { x: -0.3, z: lerp(15, 4.5, t / CYC), walk: t * 12 }, fg: [[-2.2, 3.5], [-2.2, 6.5], [-2.2, 9.5], [-2.2, 12.5]] });
  const carB = (kk, t) => world(kk, w, h, cam({ y: 1.6, pitch: 0.06 }), { city: true, mountains: false, trees: false, sun: [-26, 40, 150], extra: [[1.0, 0, lerp(17, 4.5, t / CYC), CAR[3]]] });
  let i = 0; while (u >= XC[i + 1]) i++;
  const isA = i % 2 === 0;
  blend(k, kk => (isA ? runA : carB)(kk, u), kk => runA(kk, 0), eIOs(r));   // 最后 0.3 秒回到 A 的起始状态。
  const segs = []; for (let j = 0; j < XC.length - 1; j++) segs.push([XC[j], XC[j + 1], j % 2 ? 'B' : 'A', j % 2 ? C.blue : C.g2]);
  strip(k, w, h, segs, u);
  return r > .5 || isA ? '场景 A' : '场景 B';
});
