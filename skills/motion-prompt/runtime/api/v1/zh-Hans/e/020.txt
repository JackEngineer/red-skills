// 穿越缩放 / 英文名：Zoom through
// 素材包：motion (动态图形) · 元素 20 · 类别：转场与强调
// 示例内容：镜头推进圆环的孔洞，并从中进入下一个场景。
// 动效规格：
//   - 黄底场景 X（显示“放”“大”和两个深色圆环）→ 蓝底场景 Y（显示“走”、一个白色圆环和黄色感叹号）→ 场景 X，每段 2.35 s。
//   - 当前场景以圆环孔洞的中心为缩放中心（半径 24.5）。最大缩放为孔洞恰好覆盖全画面时所需倍率的 1.08 倍，约为 14 倍。
//   - 缩放 = maxScale^p，使用指数缩放，视觉上呈匀速推进。
//   - p = 0.8 × 三次方缓入（0.25–1.2 s）+ 0.2 × 三次方缓出（1.2–1.65 s），前段加速，末段减速。
//   - 孔洞内部的下一场景以 1/maxScale 比例放置。推进结束时恰好恢复完整尺寸，随后画面切换到下一场景。
// 适用场景：适合钻入标志或字母，带着动势进入下一场景。
// 标签：缩放、穿越缩放、转场、推近、比例缩放、隧道、场景切换
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion -> ../../kit/motion.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#020
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 20
function ring(k, x, y, col) { k.strokeStyle = col; k.lineWidth = 21; k.beginPath(); k.arc(x, y, 35, 0, 7); k.stroke(); }
const ZP = [{ x: 228, y: 106, r: 24.5 }, { x: 300, y: 106, r: 24.5 }];
function scrX(k, w, h) { k.fillStyle = C.yel; k.fillRect(0, 0, w, h); const f = `800 96px ${JP}`; txt(k, '放', 128, 110, f, C.navy, 'center', 'middle'); ring(k, 228, 106, C.navy); ring(k, 324, 106, C.navy); txt(k, '大', 432, 110, f, C.navy, 'center', 'middle'); }
function scrY(k, w, h) { k.fillStyle = C.blue; k.fillRect(0, 0, w, h); const f = `800 96px ${JP}`; txt(k, '走', 208, 110, f, '#fff', 'center', 'middle'); ring(k, 300, 106, '#fff'); txt(k, '！', 372, 110, f, C.yel, 'center', 'middle'); }
const SCZ = [scrX, scrY], ZBG = [C.yel, C.blue];
CELLS[19] = abCell((k, w, h, u, a, b) => {
  const po = ZP[a], Smax = Math.hypot(Math.max(po.x, w - po.x), Math.max(po.y, h - po.y)) / po.r * 1.08;
  const p = eIn(P(u, TA - 0.25, TB - 0.3)) * 0.8 + eOut(P(u, TB - 0.3, TB + 0.15)) * 0.2;
  if (p >= 1) return SCZ[b](k, w, h);
  const s = Math.pow(Smax, p), Z = (kk, sc) => { kk.translate(po.x, po.y); kk.scale(sc, sc); kk.translate(-po.x, -po.y); };
  k.save(); Z(k, s); SCZ[a](k, w, h);
  k.save(); k.beginPath(); k.arc(po.x, po.y, po.r, 0, 7); k.clip(); k.fillStyle = ZBG[b]; k.fillRect(0, 0, w, h); Z(k, 1 / Smax); SCZ[b](k, w, h); k.restore();
  k.restore();
});
