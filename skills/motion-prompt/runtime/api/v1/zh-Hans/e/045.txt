// 动效规格：
//   - 衔接：1.7 s 至 2.8 s，历时 1.1 s；边缘从左侧画面外 x = -4 px 移动至右侧画面外 x = 543 px。
//   - 缓动：三次方缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
//   - 边缘左侧显示后画面，右侧显示前画面；两幅画面均不移动，通过裁剪揭开，而非推移。
//   - 边缘：4 px 蓝色竖线，仅在运动时可见。
//   - 相机：两幅画面均固定，主体也都不动。
//   - 每 4.7 s 在 A→B 和 B→A 间交替，往返 9.4 s；右上角显示“A → B” / “B → A”。

// 尺寸：539x208 px · 循环时长：9.4 秒

// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 21
CELLS[20] = shot(abCell((k, w, h, u, A, B) => {
  const p = eIO(P(u, 1.7, 2.8)), ex = p * (w + 8) - 4;
  A(k, w, h);
  if (p > 0) { k.save(); k.beginPath(); k.rect(0, 0, ex, h); k.clip(); k.fillStyle = '#fff'; k.fillRect(0, 0, w, h); B(k, w, h); k.restore(); }
  if (p > 0 && p < 1) { k.fillStyle = C.blue; k.fillRect(ex - 2, 0, 4, h); }
}));
