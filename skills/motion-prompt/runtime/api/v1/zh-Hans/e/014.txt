// 运动拖尾 / 英文名：Motion trail
// 素材包：motion (动态图形) · 元素 14 · 类别：运动细节
// 示例内容：小球沿“8”字路径移动，身后留下一串逐渐淡去的残影。
// 动效规格：
//   - 小球（珊瑚色，半径 25，白色中心半径 11）每 4.7 s 沿“8”字移动一周：x = 196·sin(a), y = 62·sin(2a)，a 匀速增加。
//   - 15 个残影，沿路径向后每个间隔 0.105 rad（约 0.079 s）。
//   - 残影半径从 22→6，不透明度为 0.9 × (1 − i/15) + 0.05，因此越旧越小、越淡。
//   - 残影颜色从新到旧依次为珊瑚色 → 黄色 → 蓝色。
//   - 先绘制最旧的残影，最后绘制小球本体。
// 适用场景：适合表现速度或轨迹，例如飞行物体、光标和光点。
// 标签：轨迹、运动拖尾、运动残影、残像、残影、路径、速度
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion -> ../../kit/motion.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#014
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 14
CELLS[13] = (k, w, h, u) => {
  stage(k);
  const X = w / 2, Y = h / 2 + 2, A = 196, B = 62, th = u / CYC * Math.PI * 2;
  const pt = a => [X + A * Math.sin(a), Y + B * Math.sin(2 * a)];
  const N = 15;
  for (let i = N; i >= 1; i--) {
    const [x, y] = pt(th - i * 0.105), f = i / N;
    k.globalAlpha = 0.9 * (1 - f) + 0.05; k.fillStyle = f < 0.5 ? mix(C.coral, C.yel, f * 2) : mix(C.yel, C.blue, (f - 0.5) * 2);
    k.beginPath(); k.arc(x, y, lerp(22, 6, f), 0, 7); k.fill();
  }
  k.globalAlpha = 1; const [x, y] = pt(th);
  k.fillStyle = C.coral; k.beginPath(); k.arc(x, y, 25, 0, 7); k.fill();
  k.fillStyle = '#fff'; k.beginPath(); k.arc(x, y, 11, 0, 7); k.fill();
};
