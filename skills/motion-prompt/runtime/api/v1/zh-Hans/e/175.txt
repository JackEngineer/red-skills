// 固定镜头 / 英文名：Locked-off shot
// 素材包：camera2 (镜头运动与转场（第 2 弹）) · 元素 07 · 类别：镜头运动
// 示例内容：人物从固定在三脚架上的相机前经过，汽车从后方驶过，相机一像素也不移动。
// 动效规格：
//   - 相机：完全固定（高度 1.4 m，朝向正前方，焦距 330 px）；只有主体移动。
//   - 主体：人物在前方 6.5 m 处，以恒定速度从左向右走过（4.7 s 内 x = -6.2 m → +6.2 m）；腿和手臂按 sin(u×7)×0.42 rad 摆动。
//   - 灰色汽车在 15 m 远处从右向左行驶（4.7 s 内 x = +16 m → -16 m）。
//   - 两者起点和终点均在画面外，因此每 4.7 s 无缝循环。
//   - 左下角有灰色三脚架标记；右上角读数为“机位固定”（#040 定格会停止素材时间；此处时间继续流逝，仅相机静止）。
// 适用场景：适合以静衬动突出主体、定点观察，或作为延时摄影与合成的基础。
// 标签：锁定机位、锁定机位、固定镜头、固定机位、三脚架、静止、相机固定
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 camera2 -> ../../kit/camera2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#175
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 07
CELLS[6] = shot((k, w, h, u) => {
  const q = u / CYC;
  world(k, w, h, cam({ y: 1.4 }), {
    person: { x: lerp(-6.2, 6.2, q), z: 6.5, walk: u * 7 },
    extra: [[lerp(16, -16, q), 0, 15, (kk) => { kk.fillStyle = C.g4; rr(kk, -1.9, 0.35, 3.8, 0.75, 0.25); kk.fill(); rr(kk, -1.1, 0.95, 2.1, 0.62, 0.28); kk.fill(); kk.fillStyle = C.ink; [-1.15, 1.15].forEach(x => { kk.beginPath(); kk.arc(x, 0.34, 0.34, 0, 7); kk.fill(); }); }]],
    after: (kk) => {
      kk.strokeStyle = C.g4; kk.lineWidth = 2.2; kk.lineCap = 'round'; const x = -w / 2 + 38, y = h / 2 - 20;
      kk.beginPath(); kk.moveTo(x, y - 30); kk.lineTo(x - 12, y); kk.moveTo(x, y - 30); kk.lineTo(x + 12, y); kk.moveTo(x, y - 30); kk.lineTo(x, y - 2); kk.stroke();
      kk.fillStyle = C.g4; rr(kk, x - 9, y - 42, 18, 12, 3); kk.fill();
    }
  });
  return '机位固定';
});
