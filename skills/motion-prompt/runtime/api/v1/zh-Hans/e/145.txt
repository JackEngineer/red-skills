// 滚动时间轴 / 英文名：Timeline on scroll
// 素材包：lp2 (落地页与网页效果（第 2 弹）) · 元素 01 · 类别：滚动效果
// 示例内容：页面向下滚动时，橙色线沿时间轴向下延伸；到达各节点时，圆点填实，年份和事件从右侧滑入。向上滚动则反向收回。
// 动效规格：
//   - 浏览器视口 511 × 168。滚动：0.3–2.3 s 从 0→190 px，3.0–4.3 s 从 190→0（三次方缓入缓出）。一次演示 4.7 s，每个 9.4 s 素材包周期内重复两次。
//   - 时间轴位于 x = 168：橙线沿灰色轨道（宽 3）延伸。线端位于 scroll + 126 px，因此始终距视口顶部 126 px（距底部 42 px）。
//   - 五个项目（2018–2026），圆点为白色、半径 7.5、圆环宽 2.5，位于页面 y = 72、126、180、234、288。
//   - 线端经过圆点后，再延伸 22 px 的过程中（三次方缓出）：圆环变橙，内圆半径从 0 增至 4，年份（18 px，橙色）和事件（15 px）从右侧滑入 16 px 并淡入。
//   - 反向滚动时，按相同公式收短线条并隐藏项目。右侧滚动条滑块随滚动移动。
// 适用场景：适合按时间顺序呈现公司历史、里程碑或分步流程。
// 标签：时间轴、滚动、历史、里程碑、滚动驱动、进度线
// 尺寸：539x208 px · 循环时长：9.4 秒
// 依赖：共用辅助代码 lp2 -> ../../kit/lp2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#145
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 01
CELLS[0] = (k, w, h, u) => {
  u = snapU(u);
  frame(k);
  const y = steps([[0.3, 0], [2.3, 190], [3.0, 190], [4.3, 0]], u);
  const ROWS = [[72, '2018', '创立'], [126, '2020', '首家门店开业'], [180, '2022', '会员突破一万人'], [234, '2024', '应用上线'], [288, '2026', '拓展海外']];
  const LX = 168, L = y + 126;                         // 线端随页面延伸，最远到窗口底部上方 42 像素处。
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '时间线', 34, 34, `800 11px ${EN}`, C.acc, 'left', 'alphabetic', '2px');
    txt(k, '发展历程', 34, 60, `800 20px ${JP}`, C.ink);
    pic(k, 34, 76, 100, 62, 0, 6); bars(k, 34, 148, 90, 2);
    k.fillStyle = C.faint; rr(k, LX - 1.5, 52, 3, 248, 1.5); k.fill();
    k.fillStyle = C.acc; rr(k, LX - 1.5, 52, 3, clamp(L - 52, 0, 248), 1.5); k.fill();
    for (const [iy, yr, lab] of ROWS) {
      const q = eOut(clamp((L - iy) / 22));
      k.fillStyle = '#fff'; k.beginPath(); k.arc(LX, iy, 7.5, 0, 7); k.fill();
      k.lineWidth = 2.5; k.strokeStyle = q > 0 ? C.acc : C.faint; k.stroke();
      if (q > 0) { k.fillStyle = C.acc; k.beginPath(); k.arc(LX, iy, 4 * q, 0, 7); k.fill(); }
      k.save(); k.globalAlpha = q; k.translate(16 * (1 - q), 0);
      txt(k, yr, LX + 22, iy + 6, `800 18px ${EN}`, C.acc);
      txt(k, lab, LX + 82, iy + 5, `700 15px ${JP}`, C.ink);
      bars(k, LX + 82, iy + 15, 190, 1, C.faint, 5);
      k.restore();
    }
    k.restore();
    sbar(k, y / 190, 0.45);
  });
};
