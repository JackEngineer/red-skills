// 蓝调时刻 / 英文名：Blue hour
// 素材包：photo (摄影效果、光线与风格) · 元素 08 · 类别：布光与时段
// 示例内容：太阳落山后，天空变成两层深蓝色，窗户与路灯亮起琥珀色光，人物变成深蓝色轮廓。
// 动效规格：
//   - 场景 1：地平线位于 y 150，前方为头肩人像（头部中心 x 270、y 78，半径 29 px，无五官或头发）。
//   - 太阳：半径 18 px，位于 x 120，在效果达到 70% 前从 y 40 降至 y 190，在效果 40% 至 80% 之间淡出。
//   - 颜色：天空顶部 #2C3B5E、地平线附近 #4A6191，地面 #283452，建筑 #34446B，人物 #1C2439。
//   - 灯光：每三扇窗中的一扇和路灯灯泡变为 #FFC75A；灯泡周围有半径 13 px 和 8 px 的平涂光环（#FFE3A3，不透明度 35% 和 60%）。
//   - 右上角时间 12:00 → 18:35。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合同时需要蓝色天空与明亮灯光的城市和夜景。
// 标签：蓝调时刻、暮光、暮色、城市夜景、城市景观、深蓝天空
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#104
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 08
CELLS[7] = cell((k, u) => {
  const a = env(u);
  const p = pal(BASE, NIGHT, a);
  stagePortrait(k, p, {
    sun: [120, lerp(40, 190, P(a, 0, 0.7)), 18, 1 - P(a, 0.4, 0.8)],
    after: D => { const k2 = D.k; k2.save(); k2.globalAlpha = 0.35 * a; D.circ(413, HZ - 53, 13, BASE.glow); k2.globalAlpha = 0.6 * a; D.circ(413, HZ - 53, 8, BASE.glow); k2.restore(); D.circ(413, HZ - 53, 4, p.bulb); }
  });
  bust(k, PX, PY, PR, p.person);
  hud(k, clock(lerp(720, 1115, a)));
});
