// 镜头眩光 / 英文名：Lens flare
// 素材包：photo (摄影效果、光线与风格) · 元素 13 · 类别：镜头与相机效果
// 示例内容：太阳从建筑后方露出时，周围出现光芒；一串光点、六边形和圆环沿太阳经过画面中心的连线延伸到另一侧。
// 动效规格：
//   - 场景 2：街角，建筑、树木和路灯（x 114、292、494）位于 y 132–162 的道路后方；人物站在左前方（脚部位于 x 176、y 200，身高 124 px）。
//   - 太阳：半径 13 px，从建筑后方（x 428、y 74）移动到开阔天空（x 474、y 30）。
//   - 光芒：共 8 条，从距太阳中心 17 px 处延伸至 26 px 或 44 px，交替排列（长 9 px 和 27 px；琥珀色，1.6 px，不透明度 75%）。
//   - 眩光：沿太阳经过画面中心的连线，在 t = 0.45、0.8、1.15、1.45、1.75 和 2.0 处依次放置 7 px 圆、13 px 六边形、5 px 圆、22 px 六边形、9 px 圆和 30 px 圆环（线宽 3 px）；t = 1 为画面中心。
//   - 颜色为琥珀色 #FFC75A 和 #FFE3A3，不透明度 40–70% × 效果强度。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合增加眩光、夏日阳光或电影感光效。
// 标签：镜头眩光、镜头眩光、鬼影、太阳眩光、星芒、电影感
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#109
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 13
CELLS[12] = cell((k, u) => {
  const a = env(u), sx = lerp(428, 474, a), sy = lerp(74, 30, a);
  stageStreet(k, BASE, { sun: [sx, sy, 13, 1] });
  const cx = CW / 2, cy = CH / 2, dx = cx - sx, dy = cy - sy;
  k.save();
  k.globalAlpha = 0.75 * a; k.strokeStyle = BASE.lit; k.lineWidth = 1.6; k.lineCap = 'round';
  for (let i = 0; i < 8; i++) { const t = i / 8 * TAU + 0.2, L = i % 2 ? 26 : 44; k.beginPath(); k.moveTo(sx + Math.cos(t) * 17, sy + Math.sin(t) * 17); k.lineTo(sx + Math.cos(t) * L, sy + Math.sin(t) * L); k.stroke(); }
  // 光斑沿太阳与画面中心的连线分布，t 的取值范围为 0.45 至 2。
  const G = [[0.45, 7, 'c', 0.6], [0.8, 13, 'h', 0.55], [1.15, 5, 'c', 0.7], [1.45, 22, 'h', 0.4], [1.75, 9, 'c', 0.55], [2.0, 30, 'r', 0.55]];
  for (const [t, r, kind, al] of G) {
    const x = sx + dx * t, y = sy + dy * t;
    k.globalAlpha = al * a;
    if (kind === 'r') { k.strokeStyle = BASE.lit; k.lineWidth = 3; k.beginPath(); k.arc(x, y, r, 0, TAU); k.stroke(); continue; }
    k.fillStyle = kind === 'h' ? BASE.glow : BASE.lit; k.beginPath();
    if (kind === 'h') for (let i = 0; i < 6; i++) { const t2 = i / 6 * TAU; k.lineTo(x + Math.cos(t2) * r, y + Math.sin(t2) * r); } else k.arc(x, y, r, 0, TAU);
    k.closePath(); k.fill();
  }
  k.restore();
});
