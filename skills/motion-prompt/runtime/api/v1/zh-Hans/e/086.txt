// 下划线动画 / 英文名：Underline animation
// 素材包：lp (落地页与网页效果) · 元素 14 · 类别：悬停与点击交互
// 示例内容：光标悬停的链接下方，一条橙线从左向右展开；移开后，线向右收起消失。
// 动效规格：
//   - 三个链接，20 px、粗体、相隔 56。光标从左到右各停留约 1 s（命中区域比文字左右各宽 6 px）。
//   - 下划线橙色、粗 3，位于基线下方 8 px。
//   - 悬停时，宽度以左端为起点，在 0.3 s 内从 0→100%（三次方缓出）。
//   - 离开时，左端在 0.3 s 内右移，使线向右消失（类似以右侧为 transform-origin 的 scaleX）。
// 适用场景：适合菜单或文字链接，以轻微反馈标出光标位置。
// 标签：下划线、悬停、链接、导航、菜单、文字链接、横向缩放
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#086
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 14
CELLS[13] = (k, w, h, u) => {
  frame(k);
  const LINKS = ['服务', '案例', '公司简介'], F = `800 20px ${JP}`, GAP = 56, LY = 84, DUR = 0.3;
  const cur = t => path([[0, 120, 150], [0.45, 170, 80], [1.3, 186, 84], [1.55, 268, 80], [2.4, 282, 84], [2.65, 372, 80], [3.5, 390, 84], [3.95, 430, 150], [4.8, 120, 150]], t);
  view(k, () => {
    logo(k, 22, 28); bars(k, 90, 132, 331, 2, C.faint, 6, 12, 0.6);
    k.font = F; const ws = LINKS.map(s => k.measureText(s).width); let x = (VW - ws.reduce((a, b) => a + b, 0) - GAP * (LINKS.length - 1)) / 2;
    LINKS.forEach((s, i) => {
      const x0 = x, x1 = x + ws[i], inside = t => { const [a, b] = cur(t); return a > x0 - 6 && a < x1 + 6 && b > LY - 22 && b < LY + 14; };
      txt(k, s, x0, LY, F, C.ink);
      // 光标移入和移出都用 0.3 秒。移入时下划线自左侧伸长，移出时向右缩短至消失。
      const on = inside(u); let tc = null; for (let j = 0, t = u; j < 600; j++, t -= 1 / 120) if (inside(t - 1 / 120) !== on) { tc = t; break; }
      if (tc !== null) {
        const q = eOut(P(u, tc, tc + DUR)), a = on ? 0 : q, b = on ? q : 1;
        if (b > a) { k.fillStyle = C.acc; k.fillRect(x0 + ws[i] * a, LY + 8, ws[i] * (b - a), 3); }
      }
      x += ws[i] + GAP;
    });
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
