// 波浪文字 / 英文名：Wave text
// 素材包：motion2 (动态图形（第 2 弹）) · 元素 10 · 类别：文字与图形
// 示例内容：“文字随波起伏”的汉字依次上下起伏，像波浪一样。
// 动效规格：
//   - “文字随波起伏”（70 px，粗体，字距 4 px），每 4.7 s 起伏三轮，每轮 1.567 s。
//   - 第 i 个汉字的垂直偏移 = −28·sin(φ − 0.62i)，其中 φ = 2π × 3 × 时间 ÷ 4.7，每个汉字比前一个滞后 0.62 rad。
//   - 上升中的汉字垂直拉伸 (1 + 0.06·s)，颜色从白色向黄色变化；s 为正弦值，颜色计算时限制在 0–1。
//   - 每个汉字下方的椭圆阴影随汉字上升而缩小。
// 适用场景：适合标志、短标题或随音乐出现的字幕，增加趣味和节奏感。
// 标签：波浪文字、波浪、文字、字母、交错动画、弹跳、动态文字
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion2 -> ../../kit/motion2.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#130
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 10
CELLS[9] = (k, w, h, u) => {
  stage(k);
  const f = `800 70px ${JP}`, cs = chars(k, '文字随波起伏', f, 4), Y = h / 2 + 2, ph = 2 * Math.PI * 3 * u / CYC;
  cs.forEach((c, i) => {
    if (c.ch === ' ') return;
    const s = Math.sin(ph - i * 0.62), y = -28 * s;
    k.fillStyle = 'rgba(0,0,0,.3)'; k.beginPath(); k.ellipse(w / 2 + c.x, Y + 62, c.w * (0.42 - 0.12 * s), 4, 0, 0, 7); k.fill();
    k.save(); k.translate(w / 2 + c.x, Y + y); k.scale(1, 1 + 0.06 * s); txt(k, c.ch, 0, 0, f, mix('#FFFFFF', C.yel, clamp(s)), 'center', 'middle'); k.restore();
  });
};
