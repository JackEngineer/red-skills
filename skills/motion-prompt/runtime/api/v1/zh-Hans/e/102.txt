// 蝴蝶光 / 英文名：Butterfly lighting
// 素材包：photo (摄影效果、光线与风格) · 元素 06 · 类别：布光与时段
// 示例内容：脸部前上方的灯光亮起，两侧脸部均匀受光，只有下巴下方留下一小块对称阴影。
// 动效规格：
//   - 场景 1：地平线位于 y 150，前方为头肩人像（头部中心 x 270、y 78，半径 29 px，无五官或头发）。背景变暗（天空 #E7E5E1）。
//   - 头部：整体受光，颜色 #FFC75A（没有暗侧）。
//   - 下巴下方的阴影：椭圆中心位于 y 78+1.06R，宽 0.72R、高 0.38R（R = 29，约 21×11 px），颜色 #33405A。
//   - 身体：中心 ±0.8×56 px 内受光，外侧窄带保留底色；俯视图显示灯光位于相机一侧（0°），右上角读数“0° · ↑45°”。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合突出面部美感的美妆和时尚人像。
// 标签：蝴蝶光、派拉蒙布光、美妆人像布光、人像布光、顶光、魅力人像
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#102
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 06
CELLS[5] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    torso(kk, PX, PY, R, PX - 0.8 * W, PX + 0.8 * W, lit);
    kk.fillStyle = sh; kk.beginPath(); kk.ellipse(PX, PY + R * 1.06, R * 0.72, R * 0.38, 0, 0, TAU); kk.fill();
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
  });
  plan(k, [[0, a]]);
  hud(k, '0° · ↑45°');
});
