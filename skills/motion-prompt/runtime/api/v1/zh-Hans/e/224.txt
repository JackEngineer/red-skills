// 鱼骨图 / 英文名：Fishbone diagram
// 素材包：diagram (资料中常用的图解与图表名称) · 元素 08 · 类别：整理思路的图
// 示例内容：主骨向右延伸至鱼头（结果），斜骨从上下两侧接入主骨，每根斜骨上再生出小侧骨（原因）。
// 动效规格：
//   - 主骨：尾部出现后（0.35–0.6 s），一条从 x = 52 到 x = 404 的线（深灰、3 px、头部 14 px）在 0.45 s 至 1.0 s 延伸；鱼头为写有“结果”的 92×56 px 蓝框，在 0.95 s 弹性出现
//   - 斜骨：六根斜骨从主骨的 x = 150、258 和 366 处向左上和左下倾斜（横向 62 px，向上或向下 80 px，中蓝、3 px）；从 1.1 s 开始，每列间隔 0.14 s，各用 0.35 s 绘出，下方斜骨再晚 0.07 s
//   - 侧骨：每根斜骨的 36% 和 68% 处，向左延伸一条长 26 px 的水平线（灰色、2.5 px），紧随斜骨之后、间隔 0.08 s，各用 0.22 s 长出
//   - 结束：整个图形在 4.1 s 至 4.5 s 淡出，卡片保持空白，直到 4.7 s 循环重新开始
// 适用场景：适合按类别列出导致某个结果的可能原因。
// 标签：鱼骨、鱼骨图、石川图、因果关系、根因分析、成因、质量分析
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 diagram -> ../../kit/diagram.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#224
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 08
function fishbone(k, w, h, u) {
  bg(k, w, h);
  k.globalAlpha = fade(u);
  const y = h / 2, x0 = 52, x1 = 404, hx = 452;
  const ta = easeOut(P(u, 0.35, 0.6));
  if (ta > 0) {
    k.save(); k.globalAlpha *= ta; pen(k, C.mark, 3);
    k.beginPath(); k.moveTo(x0 - 26, y - 22); k.lineTo(x0, y); k.lineTo(x0 - 26, y + 22); k.stroke(); k.restore();
  }
  const sp = easeInOut(P(u, 0.45, 1.0));
  if (sp > 0) { pen(k, C.mark, 3); trace(k, [[x0, y], [x1, y]], sp, 14, C.mark); }
  const hs = pop(u, 0.95);
  if (hs > 0.01) {
    k.save(); k.translate(hx, y); k.scale(hs, hs);
    k.fillStyle = C.blue; rr(k, -38, -28, 92, 56, 12); k.fill();
    txt(k, '结果', 8, 1, `700 20px ${FONT}`, '#FFFFFF', 'center', 'middle');
    k.restore();
  }
  // 主鱼骨从主轴上的 bx 处连向左上或左下方。
  const bones = [[150, -1], [258, -1], [366, -1], [150, 1], [258, 1], [366, 1]];
  bones.forEach(([bx, sd], i) => {
    const a = 1.1 + (i % 3) * 0.14 + (sd > 0 ? 0.07 : 0);
    const t = easeInOut(P(u, a, a + 0.35));
    if (t <= 0) return;
    const tip = [bx - 62, y + sd * 80];
    pen(k, C.blueM, 3); trace(k, [[bx, y], tip], t);
    // 两条次鱼骨在主鱼骨的三分之一和三分之二处水平向左伸出。
    [0.36, 0.68].forEach((f, j) => {
      const st = easeOut(P(u, a + 0.38 + j * 0.08, a + 0.6 + j * 0.08));
      if (st <= 0) return;
      const px = lerp(tip[0], bx, f), py = lerp(tip[1], y, f);
      pen(k, C.g4, 2.5); k.beginPath(); k.moveTo(px, py); k.lineTo(px - 26 * st, py); k.stroke();
    });
  });
  k.globalAlpha = 1;
}
CELLS[7] = fishbone;
