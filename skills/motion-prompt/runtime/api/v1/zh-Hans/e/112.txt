// 示例内容：画面从四角向内平滑变暗，中间的人物和道路保持明亮。
// 动效规格：
//   - 场景 2：街角，建筑、树木和路灯（x 114、292、494）位于 y 132–162 的道路后方；人物站在左前方（脚部位于 x 176、y 200，身高 124 px）。
//   - 形状：以画面中心为中心的椭圆（纵向半径 = 横向半径 × 208/539 × 1.35）。
//   - 横向半径为 334 px，内侧 45% 区域完全透明，之后 rgb(32,35,43) 朝边缘从不透明度 0% 增至 70%。
//   - 时间：效果在 0.4 s 至 1.4 s 之间显现，保持至 4.0 s，并在 4.8 s 前消失（单次 = 5 s，三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)）。
// 适用场景：适合将注意力集中到主体，或表现老镜头效果。
// 标签：暗角、边缘变暗、暗角、边缘变暗、边缘失光、焦点
// 尺寸：539x208 px · 循环时长：5 秒
// 依赖：共用辅助代码 photo -> ../../kit/photo.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#112
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 16
CELLS[15] = cell((k, u) => {
  const a = env(u);
  stageStreet(k, BASE);
  // 椭圆径向渐变在半径的 45% 以内完全透明，向外增加到 70% 不透明度。
  k.save(); k.translate(CW / 2, CH / 2); k.scale(1, CH / CW * 1.35);
  const g = k.createRadialGradient(0, 0, CW * 0.45 * 0.62, 0, 0, CW * 0.62);
  g.addColorStop(0, 'rgba(32,35,43,0)'); g.addColorStop(1, `rgba(32,35,43,${(0.7 * a).toFixed(3)})`);
  k.fillStyle = g; k.fillRect(-CW, -CW, CW * 2, CW * 2); k.restore();
});
