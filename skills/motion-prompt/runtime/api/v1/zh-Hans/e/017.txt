// 素材包：motion (动态图形) · 元素 17 · 类别：转场与强调
// 示例内容：圆形或星形窗口逐渐扩大，内部显露下一个场景。
// 动效规格：
//   - 白天场景 → 夜晚场景 → 白天场景，每段 2.35 s。每半段的切换在 0.45–1.6 s 进行，历时 1.15 s，使用三次方缓入缓出。
//   - 下一场景绘制在当前场景上方，并裁剪为指定形状；形状不断扩大，直到覆盖画面。
//   - 白天→夜晚：圆形从太阳中心（112, 104）扩大，半径 0→470。
//   - 夜晚→白天：五角星从月亮中心（427, 104）扩大，外半径 0→1150，扩大的同时旋转 1.2 rad（约 69°）。
//   - 扩大时，形状边缘带白色描边，线宽 7。
// 适用场景：适合从画面中的太阳、标志或按钮等物体处开始转场。
// 标签：遮罩、转场、圆形光圈、划像转场、图形划像、显现、场景切换
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 motion -> ../../kit/motion.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#017
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 17
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
