// 折叠面板 / 英文名：Accordion
// 素材包：ui (UI 组件) · 元素 04 · 类别：浮层与展开
// 示例内容：点击三个标题中的第一个，展开其内容并将下方各行向下推；再次点击收起。
// 动效规格：
//   - 542×191 卡片，循环周期 4.5 s。三个标题宽 494、高 34、圆角 8、间距 6，右侧带向下箭头。
//   - 光标在 0.95 s 到达第一个标题，在 1.0 s 和 3.3 s 点击。
//   - 展开量 ex：1.0–1.35 s 以 easeInOutCubic 从 0→1，3.3–3.62 s 从 1→0。
//   - 第一行高度从 34 增至 92（+58·ex），将下面的行向下推；两行正文裁切在行内，以不透明度 ex 淡入。
//   - 展开时，箭头旋转 180°·ex，边框变浅蓝色，标题变蓝色。
// 适用场景：适合常见问题或较长的详情，让用户浏览标题后只展开需要的内容。
// 标签：折叠面板、可折叠、展开、收起、展开与收起、常见问题、细节
// 尺寸：542x191 px · 循环时长：4.5 秒
// 依赖：共用辅助代码 ui -> ../../kit/ui.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#052
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 04
CELLS[3] = ui((w, h, t) => {
  const ex = eio(seg(t, 1.0, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
  let y = 12; const H0 = 34, EX = 58;
  for (let i = 0; i < 3; i++) {
    const extra = i === 0 ? EX * ex : 0;
    strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
    bar(42, y + 12, [180, 220, 150][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
    chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : '#9a9896');
    if (i === 0 && ex > 0) { X.save(); rr(24, y, 494, H0 + extra, 8); X.clip(); X.globalAlpha = ex; bar(42, y + H0 + 6, 420); bar(42, y + H0 + 24, 320); X.restore(); }
    y += H0 + extra + 6;
  }
  pointer([[.3, w + 30, h + 30], [.95, 300, 31], [1.6, 300, 31], [3.0, 300, 31], [3.55, 300, 31], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
