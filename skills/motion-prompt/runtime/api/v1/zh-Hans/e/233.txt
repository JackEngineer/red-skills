// 甘特图 / 英文名：Gantt chart
// 素材包：diagram (资料中常用的图解与图表名称) · 元素 17 · 类别：展示流程与全局的图
// 示例内容：顶部为日期刻度（1–8），左侧为任务 A–E；各条形从错开的列位置起，依次向右生长。
// 动效规格：
//   - 刻度：顶部横排日期 1–8（列宽 52 px），带浅色竖线，整个循环中保留
//   - 任务：左侧五行 A–E，行距 27 px；条形高 20 px、圆角 6 px；起止位置为 A 0–2、B 1.5–3.5、C 3–4.5、D 3.5–6.5、E 6–8（以列为单位）
//   - 运动：从 0.4 s 开始，间隔 0.3 s，每条从左端向右用 0.5 s 生长（ease-out）
//   - 颜色：D 为深蓝色，其余为中蓝色
//   - 结束：整个图形在 4.1 s 至 4.5 s 淡出，卡片保持空白，直到 4.7 s 循环重新开始
// 适用场景：适合展示项目中各项任务的时长，以及任务之间的重叠和先后关系。
// 标签：甘特、甘特图、日程安排、项目计划、时间轴条形、任务、路线图
// 尺寸：539x208 px · 循环时长：4.7 秒
// 依赖：共用辅助代码 diagram -> ../../kit/diagram.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#233
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 17
function gantt(k, w, h, u) {
  bg(k, w, h);
  const x0 = 84, cols = 8, cw = 52, yAxis = 44, f = fade(u);
  for (let i = 0; i < cols; i++) txt(k, String(i + 1), x0 + cw * i + cw / 2, 34, `500 18px ${FONT}`, C.sub, 'center');
  k.strokeStyle = C.g2; k.lineWidth = 2;
  for (let i = 0; i <= cols; i++) { k.beginPath(); k.moveTo(x0 + cw * i, yAxis); k.lineTo(x0 + cw * i, 194); k.stroke(); }
  k.strokeStyle = C.g4; k.lineWidth = 2.5; k.lineCap = 'round';
  k.beginPath(); k.moveTo(x0, yAxis); k.lineTo(x0 + cw * cols, yAxis); k.stroke();
  for (let i = 0; i <= cols; i++) { k.beginPath(); k.moveTo(x0 + cw * i, yAxis - 6); k.lineTo(x0 + cw * i, yAxis); k.stroke(); }
  const T = [[0, 2], [1.5, 2], [3, 1.5], [3.5, 3], [6, 2]];
  T.forEach(([s, len], i) => {
    const y = 62 + i * 27;
    txt(k, 'ABCDE'[i], 54, y + 1, `600 18px ${FONT}`, C.g4, 'center', 'middle');
    const p = appear(u, 0.4 + i * 0.3, 0.5) * f;
    if (p <= 0.001) return;
    k.globalAlpha = Math.min(1, p * 3) * f;
    k.fillStyle = i === 3 ? C.blue : C.blueM;
    rr(k, x0 + cw * s + 3, y - 10, Math.max(6, (cw * len - 6) * appear(u, 0.4 + i * 0.3, 0.5)), 20, 6); k.fill();
    k.globalAlpha = 1;
  });
}
CELLS[16] = gantt;
