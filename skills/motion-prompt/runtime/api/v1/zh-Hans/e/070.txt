// 加载转圈 / 英文名：Spinner
// 素材包：ui (UI 组件) · 元素 22 · 类别：导航与状态反馈
// 示例内容：点击“提交”，卡片蒙上白色遮罩，中央蓝色圆弧旋转；完成后变成勾号并淡出。
// 动效规格：
//   - 542×191 卡片，循环周期 4.5 s。光标在 1.0 s 点击“提交”（120×40，底部中央）。
//   - 1.05–1.2 s 内（线性），整卡淡入白色遮罩（不透明度 0.78）和加载指示器。
//   - 加载指示器位于 (271, 92)：浅色圆环，半径 26、粗 6，其上有蓝色 243°（1.35π）圆弧，端点为圆头，以每秒 1.3 圈（468°/s）旋转。
//   - 2.95–3.05 s 加载指示器淡出；2.95–3.2 s 蓝色圆（半径 28，内部白色勾号）从 0.7 → 1 放大（easeOutCubic）。
//   - 所有内容在 3.75–4.0 s 线性淡出。
// 适用场景：适合发送或保存等无法确定准确时长的短暂等待。
// 标签：加载转圈、加载、加载指示器、加载指示器、环形进度、正在处理
// 尺寸：542x191 px · 循环时长：4.5 秒
// 依赖：共用辅助代码 ui -> ../../kit/ui.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#070
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 22
CELLS[21] = ui((w, h, t) => {
  const load = seg(t, 1.05, 1.2) * (1 - seg(t, 2.95, 3.05));
  const done = seg(t, 2.95, 3.15) * (1 - seg(t, 3.75, 4.0));
  pageLines(24, 18, 494); strokeRR(24, 90, 494, 34, 7, C.line2, 1.5);
  fillRR(w / 2 - 60, 140, 120, 40, 8, C.blue); text('提交', w / 2, 160, 16, '#fff', 600, 'center');
  const ov = Math.max(load, done);
  if (ov > 0) {
    X.fillStyle = `rgba(255,255,255,${.78 * ov})`; X.fillRect(0, 0, w, h);
    if (load > 0) { X.save(); X.globalAlpha = load; ring(w / 2, 92, 26, '#e3e8f5', 6); X.beginPath(); const a0 = t * Math.PI * 2 * 1.3; X.arc(w / 2, 92, 26, a0, a0 + Math.PI * 1.35); X.strokeStyle = C.blue; X.lineWidth = 6; X.lineCap = 'round'; X.stroke(); X.restore(); }
    if (done > 0) { X.save(); X.globalAlpha = done; const s = .7 + .3 * eo(seg(t, 2.95, 3.2)); X.translate(w / 2, 92); X.scale(s, s); circle(0, 0, 28, C.blue); check(0, 0, 11, '#fff', 4); X.restore(); }
  }
  pointer([[.3, w + 30, h + 30], [.95, w / 2 + 6, 166], [1.5, w / 2 + 6, 166], [2.2, w + 30, h + 30]], [1.0], t);
});
