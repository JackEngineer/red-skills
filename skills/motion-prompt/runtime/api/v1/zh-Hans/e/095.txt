// 动态渐变 / 英文名：Animated gradient
// 素材包：lp (落地页与网页效果) · 元素 23 · 类别：加载、导航与循环动效
// 示例内容：标题后方，橙色、浅橙色、蓝色和浅蓝色的柔和色块沿圆周漂移，持续混合出不同效果。
// 动效规格：
//   - 底色 #FFF1EB，上叠四个径向渐变（中心有色，边缘透明，各自不透明度 0.85）。
//   - 橙色（半径 230）、浅橙色 #FFB08F（210）、蓝色（240）、浅蓝色 #9DB6FB（200）。
//   - 每个中心每 4.8 s 沿椭圆移动一圈（x 半径 70、60、80、50；y 半径 30、40、30、35；起始角 0、2.1、4.2、1.0 rad）。
//   - 白色标题（34 px）和按钮固定在上层，只有渐变移动。
// 适用场景：适合不使用照片、又希望为首屏或标题背景增加色彩和动感的场景。
// 标签：渐变、动态渐变、网格渐变、背景、首屏主视觉、氛围、循环
// 尺寸：539x208 px · 循环时长：4.8 秒
// 依赖：共用辅助代码 lp -> ../../kit/lp.js
// 动态示例：https://jackengineer.github.io/motion-prompt/?lang=zh-Hans#095
// 以上资源路径相对于本规格文件。
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D（调用方已清空画布并用 save/restore 包裹），w、h 为尺寸，u 为秒数，范围 [0, cycle)。画面只取决于 u。
// 以下为元素代码。CELLS、常量和辅助函数在 kit 中。

// @element 23
CELLS[22] = (k, w, h, u) => {
  frame(k);
  const a = 2 * Math.PI * u / CYC;
  const BLOBS = [[C.acc, 150, 60, 230, 70, 30, 0], ['#FFB08F', 380, 40, 210, 60, 40, 2.1], [C.blue, 330, 150, 240, 80, 30, 4.2], ['#9DB6FB', 90, 170, 200, 50, 35, 1.0]];
  view(k, () => {
    k.fillStyle = '#FFF1EB'; k.fillRect(0, 0, VW, VH);
    for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
      const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
      const g = k.createRadialGradient(x, y, 0, x, y, r); g.addColorStop(0, col); g.addColorStop(1, 'rgba(255,255,255,0)');
      k.globalAlpha = 0.85; k.fillStyle = g; k.fillRect(0, 0, VW, VH); k.globalAlpha = 1;
    }
    txt(k, '你好，未来。', 40, 82, `800 34px ${EN}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 40, 100, 200, 2, 'rgba(255,255,255,.8)', 6, 12, 0.6);
    btn(k, 40, 128, 104, 26, '开始使用', '#fff', C.ink);
  });
};
